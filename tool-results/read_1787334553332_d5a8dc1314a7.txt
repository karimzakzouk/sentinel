    30→    30→// validateFilePath validates a file path returned by the LLM for a code fix.
    31→    31→// Returns the cleaned path on success, or "" if rejected. Callers should
    32→    32→// skip files with empty return values (do not abort the whole proposal — one
    33→    33→// bad path shouldn't kill a fix that has other valid files).
    34→    34→//
    35→    35→// Rules:
    36→    36→//  1. Reject empty paths
    37→    37→//  2. Reject absolute paths (anything starting with '/')
    38→    38→//  3. Reject paths containing '..' segments (path traversal)
    39→    39→//  4. Reject paths that escape the repo root after cleaning
    40→    40→//  5. Reject sensitive paths by default:
    41→    41→//     - .github/workflows/*, .github/actions/* (CI/CD tampering)
    42→    42→//     - .gitlab-ci.yml, .circleci/, Jenkinsfile, .drone.yml, cloudbuild.*, azure-pipelines.yml
    43→    43→//     - Dockerfile, docker-compose.* (build-chain tampering)
    44→    44→//     - package.json, package-lock.json (postinstall hooks)
    45→    45→//     - pyproject.toml, setup.py, requirements.txt, Pipfile, Pipfile.lock
    46→    46→//     - go.mod, go.sum
    47→    47→//     - .npmrc, .pypirc, .netrc (registry credentials)
    48→    48→//     - .env (but allow .env.example)
    49→    49→//  6. Reject paths with control characters or null bytes
    50→    50→//  7. Cap path length at 500 chars
    51→    51→func validateFilePath(path string) string {
    52→    52→        if path == "" {
    53→    53→                return ""
    54→    54→        }
    55→    55→        if len(path) > 500 {
    56→    56→                logging.Info("rejecting path (too long)", "path", path)
    57→    57→                return ""
    58→    58→        }
    59→    59→
    60→    60→        for _, r := range path {
    61→    61→                if r < 0x20 || r == 0x7f {
    62→    62→                        logging.Info("rejecting path (control char)", "path", path)
    63→    63→                        return ""
    64→    64→                }
    65→    65→        }
    66→    66→
    67→    67→        if filepath.IsAbs(path) {
    68→    68→                logging.Info("rejecting absolute path", "path", path)
    69→    69→                return ""
    70→    70→        }
    71→    71→
    72→    72→        clean := filepath.Clean(path)
    73→    73→        if clean == "." {
    74→    74→                logging.Info("rejecting path (resolves to repo root)", "path", path)
    75→    75→                return ""
    76→    76→        }
    77→    77→        if strings.HasPrefix(clean, "..") {
    78→    78→                logging.Info("rejecting path traversal", "path", path)
    79→    79→                return ""
    80→    80→        }
    81→    81→
    82→    82→        cleanSlash := filepath.ToSlash(clean)
    83→    83→        sensitive := false
    84→    84→        sensitiveReason := ""
    85→    85→        switch {
    86→    86→        case strings.HasPrefix(cleanSlash, ".github/workflows/"),
    87→    87→                strings.HasPrefix(cleanSlash, ".github/actions/"),
    88→    88→                strings.HasPrefix(cleanSlash, ".gitlab-ci.yml"),
    89→    89→                strings.HasPrefix(cleanSlash, ".circleci/"),
    90→    90→                strings.HasPrefix(cleanSlash, "Jenkinsfile"),
    91→    91→                strings.HasPrefix(cleanSlash, ".drone.yml"),
    92→    92→                strings.HasPrefix(cleanSlash, "cloudbuild.yaml"),
    93→    93→                strings.HasPrefix(cleanSlash, "cloudbuild.yml"),
    94→    94→                strings.HasPrefix(cleanSlash, ".azure-pipelines.yml"),
    95→    95→                strings.HasPrefix(cleanSlash, "azure-pipelines.yml"):
    96→    96→                sensitive = true
    97→    97→                sensitiveReason = "CI/CD-sensitive path"
    98→    98→
    99→    99→        case cleanSlash == "Dockerfile" || strings.HasPrefix(cleanSlash, "Dockerfile."),
   100→   100→                strings.HasPrefix(cleanSlash, "docker-compose.yml"),
   101→   101→                strings.HasPrefix(cleanSlash, "docker-compose.yaml"),
   102→   102→                cleanSlash == "package.json" || cleanSlash == "package-lock.json",
   103→   103→                cleanSlash == "pyproject.toml" || cleanSlash == "setup.py" || cleanSlash == "requirements.txt" || cleanSlash == "Pipfile" || cleanSlash == "Pipfile.lock",
   104→   104→                cleanSlash == "go.mod" || cleanSlash == "go.sum",
   105→   105→                cleanSlash == ".npmrc" || cleanSlash == ".pypirc" || cleanSlash == ".netrc",
   106→   106→                cleanSlash == ".env" && cleanSlash != ".env.example":
   107→   107→                sensitive = true
   108→   108→                sensitiveReason = "build-chain-sensitive path"
   109→   109→        }
   110→   110→        if sensitive {
   111→   111→                logging.Info("rejecting sensitive path", "reason", sensitiveReason, "path", path)
   112→   112→                return ""
   113→   113→        }
   114→   114→
   115→   115→        return clean
   116→   116→}

   118→   118→const fixProposerSystemPrompt = `You are Sentinel's Fix Proposer agent. You write production code fixes for runtime errors.
   119→   119→
   120→   120→Given a confirmed root cause and the source code, propose ONE fix using SEARCH/REPLACE blocks.
   121→   121→
   122→   122→## Fix Quality Requirements
   123→   123→- Fix the ROOT CAUSE, not just the symptom. If a collection access crashes (dict/key lookup, array index, map get), add a proper check — don't just wrap it in try/catch.
   124→   124→- Use proper error handling for the framework + language in use. For web APIs (FastAPI/Flask in Python, Express/Koa in JS/TS, net/http or chi/gin in Go, Spring in Java, Actix in Rust, etc.):
   125→   125→  - Bad input (missing key, invalid type, out of range) → return appropriate HTTP error (400, 404, 422)
   126→   126→  - Use the framework's error response mechanism (e.g. raise HTTPException in FastAPI, abort() in Flask, c.JSON(status, err) in gin, ResponseEntity.status(status).body(err) in Spring)
   127→   127→  - NEVER return error states with HTTP 200 — callers should know something went wrong from the status code
   128→   128→- For non-web code (CLIs, workers, daemons), use the language's idiomatic error pattern (Python: raise ValueError; Go: fmt.Errorf + return; JS: throw new Error; Java: throw new IllegalArgumentException; Rust: Result::Err).
   129→   129→- Minimal changes. Only modify what's broken. Don't refactor surrounding code.
   130→   130→- The fix must be syntactically valid in the target language (the detected language is included in the context).
   131→   131→
   132→   132→## CRITICAL: Replacement Completeness
   133→   133→After your search/replace block is applied, the resulting code must be complete and correct.
   134→   134→Before submitting, mentally apply your replacement to the search text and verify:
   135→   135→- Every variable used after the replaced block is assigned on ALL code paths (not just inside an if/try block)
   136→   136→- Control flow is complete — if you add a guard clause (if/raise, if/return, if/throw), the code AFTER the guard must still execute correctly when the guard condition is FALSE
   137→   137→- Indentation of your replacement matches the surrounding code — the replaced text must fit seamlessly at the same nesting level
   138→   138→- The function still returns/exits correctly on all paths — no code becomes unreachable, no variables become undefined

   230→   230→// Run executes the fix proposer agent.
   231→   231→func (a *FixProposer) Run(ctx context.Context, state *schemas.AgentState) error {
   232→   232→        if state.RootCause == nil {
   233→   233→                state.Status = schemas.StatusFailed
   234→   234→                s := "fix_proposer_no_root_cause"
   235→   235→                state.Error = &s
   236→   236→                return nil
   237→   237→        }
   238→   238→
   239→   239→        rc := state.RootCause
   240→   240→
   241→   241→        // Fallback: if we have no source code, propose restart. Don't ask the
   242→   242→        // LLM to write code it can't see — it'll hallucinate.
   243→   243→        if len(state.FetchedCode) == 0 {
   244→   244→                logging.Info("no source code fetched — proposing restart fallback")
   245→   245→                proposal := &schemas.FixProposal{
   246→   246→                        FixID:              uuid.New(),
   247→   247→                        FixType:            schemas.FixRestart,
   248→   248→                        TargetService:      state.Cluster.Service,
   249→   249→                        TargetNamespace:    state.Cluster.Namespace,
   250→   250→                        Description:        fmt.Sprintf("Restart fallback: no source code available to generate a code fix. Root cause: %s", rc.Statement),
   251→   251→                        Repo:               strPtr(discoverRepo(state, a.gitHost)),
   252→   252→                        PRTitle:            strPtr(fmt.Sprintf("[Sentinel] Restart %s (no source available for code fix)", state.Cluster.Service)),
   253→   253→                        PRBody:             strPtr(buildPRBody(state, rc, "restart")),
   254→   254→                        RequiresSandbox:    false,
   255→   255→                        ProposedConfidence: rc.Confidence * 0.5, // lower confidence — we're guessing
   256→   256→                }
   257→   257→                a.finalizeProposal(state, proposal)
   258→   258→                return nil
   259→   259→        }
   260→   260→
   261→   261→        // Schema migrations block autonomous rollback — force roll_forward.
   262→   262→        if rc.HasSchemaMigration && (rc.Category == "bad_deploy" || rc.Category == "schema_migration") {
   263→   263→                proposal := &schemas.FixProposal{
   264→   264→                        FixID:              uuid.New(),
   265→   265→                        FixType:            schemas.FixRollForward,
   266→   266→                        TargetService:      state.Cluster.Service,
   267→   267→                        TargetNamespace:    state.Cluster.Namespace,
   268→   268→                        Description:        fmt.Sprintf("Roll-forward fix for %s (schema migration blocks autonomous rollback)", rc.Category),
   269→   269→                        Repo:               strPtr(discoverRepo(state, a.gitHost)),
   270→   270→                        PRTitle:            strPtr(fmt.Sprintf("[Sentinel] Roll-forward fix for %s in %s", rc.Category, state.Cluster.Service)),
   271→   271→                        PRBody:             strPtr(buildPRBody(state, rc, "roll_forward")),
   272→   272→                        RequiresSandbox:    true,
   273→   273→                        ProposedConfidence: rc.Confidence,
   274→   274→                }
   275→   275→                // Still call the LLM to generate the actual code fix.
   276→   276→                a.llmCodeFix(ctx, state, proposal)
   277→   277→                a.finalizeProposal(state, proposal)
   278→   278→                return nil
   279→   279→        }
   280→   280→
   281→   281→        // Ask the LLM for the fix — with adaptive retry loop. The LLM outputs
   282→   282→        // metadata JSON (small) + SEARCH/REPLACE blocks (plain text). We parse
   283→   283→        // the blocks, apply them with layered matching (exact → fuzzy), and feed
   284→   284→        // actionable errors back on failure (Aider's pattern).
   285→   285→        userMsg := a.buildUserMessage(state)
   286→   286→        var files []schemas.FileChange
   287→   287→        var fixType schemas.FixType
   288→   288→        var dockerfilePath, prTitle, prBody, description string
   289→   289→        var lastError string
   290→   290→
   291→   291→        maxRetries := 5
   292→   292→        for attempt := 0; attempt <= maxRetries; attempt++ {
   293→   293→                var retryHint string
   294→   294→                if attempt > 0 {
   295→   295→                        retryHint = fmt.Sprintf("\n\n## PREVIOUS ATTEMPT FAILED\n%s\n\nPlease try again. Fix the issue above and return a corrected response.", lastError)
   296→   296→                        userMsg = a.buildUserMessage(state) + retryHint
   297→   297→                }
   298→   298→
   299→   299→                // Per-attempt timeout: large reasoning models can take 3-5 minutes
   300→   300→                // to generate code; the per-call cap is 300s, clipped by ctx.
   301→   301→                attemptCtx, attemptCancel := context.WithTimeout(ctx, 300*time.Second)
   302→   302→                logging.Info("fix proposer attempt", "attempt", attempt+1, "max", maxRetries+1)
   303→   303→                rawOutput, llmErr := a.router.FrontierText(attemptCtx, fixProposerSystemPrompt, userMsg, 8000)
   304→   304→                attemptCancel()
   305→   305→                if llmErr != nil {
   306→   306→                        lastError = fmt.Sprintf("LLM call failed: %v", llmErr)
   307→   307→                        logging.Error("attempt LLM error", "attempt", attempt+1, "err", llmErr)
   308→   308→                        continue
   309→   309→                }
   310→   310→
   311→   311→                logging.Info("got LLM output", "attempt", attempt+1, "bytes", len(rawOutput))
   312→   312→
   313→   313→                // Parse metadata JSON from the first line(s). The LLM outputs
   314→   314→                // {"fix_type":"code_fix",...} before the SEARCH/REPLACE blocks.
   315→   315→                metadata := extractMetadataJSON(rawOutput)
   316→   316→                fixTypeStr, _ := metadata["fix_type"].(string)
   317→   317→                if fixTypeStr == "" {
   318→   318→                        fixTypeStr = "code_fix"
   319→   319→                }
   320→   320→                fixType = schemas.FixType(fixTypeStr)
   321→   321→                if !isValidFixType(fixTypeStr) {
   322→   322→                        fixType = schemas.FixCodeFix
   323→   323→                }
   324→   324→                dockerfilePath, _ = metadata["dockerfile_path"].(string)
   325→   325→                if dockerfilePath == "" {
   326→   326→                        dockerfilePath = "Dockerfile"
   327→   327→                }
   328→   328→                prTitle, _ = metadata["pr_title"].(string)
   329→   329→                if prTitle == "" {
   330→   330→                        prTitle = fmt.Sprintf("[Sentinel] %s for %s", fixType, state.Cluster.Service)
   331→   331→                }
   332→   332→                prBody, _ = metadata["pr_body"].(string)
   333→   333→                if prBody == "" {
   334→   334→                        prBody = buildPRBody(state, rc, string(fixType))
   335→   335→                }
   336→   336→                description, _ = metadata["description"].(string)
   337→   337→                if description == "" {
   338→   338→                        description = fmt.Sprintf("Fix %s in %s", fixType, state.Cluster.Service)
   339→   339→                }
   340→   340→
   341→   341→                // Parse SEARCH/REPLACE blocks from the output.
   342→   342→                blocks := ParseEditBlocks(rawOutput)
   343→   343→
   344→   344→                if len(blocks) > 0 {
   345→   345→                        logging.Info("parsed SEARCH/REPLACE blocks", "attempt", attempt+1, "count", len(blocks))
   346→   346→
   347→   347→                        // Group blocks by file path.
   348→   348→                        blocksByFile := make(map[string][]EditBlock)
   349→   349→                        for _, b := range blocks {
   350→   350→                                path := b.FilePath
   351→   351→                                if path == "" {
   352→   352→                                        // Use the first fetched file as default.
   353→   353→                                        for p := range state.FetchedCode {
   354→   354→                                                path = p
   355→   355→                                                break
   356→   356→                                        }
   357→   357→                                }
   358→   358→                                path = normalizeFilePath(path, state.FetchedCode)
   359→   359→                                blocksByFile[path] = append(blocksByFile[path], EditBlock{
   360→   360→                                        FilePath: path,
   361→   361→                                        Search:   b.Search,
   362→   362→                                        Replace:  b.Replace,
   363→   363→                                })
   364→   364→                        }
   365→   365→
   366→   366→                        // Apply blocks to each file.
   367→   367→                        allApplied := true
   368→   368→                        files = []schemas.FileChange{}
   369→   369→                        for filePath, fileBlocks := range blocksByFile {
   370→   370→                                originalContent, ok := state.FetchedCode[filePath]
   371→   371→                                if !ok {
   372→   372→                                        lastError = fmt.Sprintf("file %s not found in fetched source code. Available files: %v", filePath, keysOf(state.FetchedCode))
   373→   373→                                        allApplied = false
   374→   374→                                        break
   375→   375→                                }
   376→   376→
   377→   377→                                editResult := ApplyEditBlocks(originalContent, fileBlocks)
   378→   378→                                if !editResult.Success {
   379→   379→                                        lastError = editResult.FailedReason
   380→   380→                                        allApplied = false
   381→   381→                                        break
   382→   382→                                }
   383→   383→
   384→   384→                                // Syntax-validate the patched content BEFORE accepting it.
   385→   385→                                // This catches broken code that the LLM generated (missing
   386→   386→                                // brackets, invalid syntax, etc.) and feeds the error back
   387→   387→                                // on the next retry. Based on the Aider/SWE-agent pattern
   388→   388→                                // of validate-then-retry.
   389→   389→                                if syntaxErr := syntaxvalidate.Validate(filePath, editResult.NewContent); syntaxErr != nil {
   390→   390→                                        lastError = fmt.Sprintf("syntax validation failed for %s: %v", filePath, syntaxErr)
   391→   391→                                        logging.Error("fix proposer: syntax validation failed", "attempt", attempt+1, "path", filePath, "err", syntaxErr)
   392→   392→                                        allApplied = false
   393→   393→                                        break
   394→   394→                                }
   395→   395→
   396→   396→                                // Build FileChange from the result — use Changes (search/replace)
   397→   397→                                // so the GitHub executor can apply them to the branch.
   398→   398→                                var changes []schemas.SearchReplace
   399→   399→                                for _, b := range fileBlocks {
   400→   400→                                        changes = append(changes, schemas.SearchReplace{
   401→   401→                                                Search:  strings.TrimRight(b.Search, "\n"),
   402→   402→                                                Replace: strings.TrimRight(b.Replace, "\n"),
   403→   403→                                        })
   404→   404→                                }
   405→   405→                                files = append(files, schemas.FileChange{
   406→   406→                                        Path:    filePath,
   407→   407→                                        Changes: changes,
   408→   408→                                })
   409→   409→                                logging.Info("applied blocks to file", "attempt", attempt+1, "blocks", len(fileBlocks), "path", filePath)
   410→   410→                        }
   411→   411→
   412→   412→                        if allApplied && len(files) > 0 {
   413→   413→                                logging.Info("fix attempt succeeded", "attempt", attempt+1, "files", len(files), "blocks", len(blocks))
   414→   414→                                break
   415→   415→                        }
   416→   416→                        logging.Error("fix attempt failed", "attempt", attempt+1, "err", lastError)
   417→   417→                        continue
   418→   418→                }
   419→   419→
   420→   420→                // No SEARCH/REPLACE blocks found — try whole-file replacement fallback.
   421→   421→                if filePath, content := ParseWholeFileReplacement(rawOutput); filePath != "" && content != "" {
   422→   422→                        filePath = normalizeFilePath(filePath, state.FetchedCode)
   423→   423→                        logging.Info("using whole-file replacement", "attempt", attempt+1, "path", filePath, "bytes", len(content))
   424→   424→                        files = []schemas.FileChange{{Path: filePath, NewContent: content}}
   425→   425→                        break
   426→   426→                }
   427→   427→
   428→   428→                lastError = "no SEARCH/REPLACE blocks or whole-file replacement found in LLM output"
   429→   429→                logging.Error("fix attempt failed", "attempt", attempt+1, "err", lastError)
   430→   430→        }
   431→   431→
   432→   432→        if len(files) == 0 {
   433→   433→                logging.Error("all fix attempts failed", "attempts", maxRetries+1)
   434→   434→                state.Status = schemas.StatusFailed
   435→   435→                s := fmt.Sprintf("fix_proposer_no_files_after_%d_attempts: %s", maxRetries+1, lastError)
   436→   436→                state.Error = &s
   437→   437→                return nil
   438→   438→        }
   439→   439→
   440→   440→        if prBody == "" {
   441→   441→                prBody = buildPRBody(state, rc, string(fixType))
   442→   442→        }
   443→   443→
   444→   444→        proposal := &schemas.FixProposal{
   445→   445→                FixID:              uuid.New(),
   446→   446→                FixType:            fixType,
   447→   447→                TargetService:      state.Cluster.Service,
   448→   448→                TargetNamespace:    state.Cluster.Namespace,
   449→   449→                Description:        description,
   450→   450→                Repo:               strPtr(discoverRepo(state, a.gitHost)),
   451→   451→                FilesToChange:      files,
   452→   452→                DockerfilePath:     strPtr(dockerfilePath),
   453→   453→                PRTitle:            strPtr(prTitle),
   454→   454→                PRBody:             strPtr(prBody),
   455→   455→                RequiresSandbox:    true,
   456→   456→                ProposedConfidence: rc.Confidence,
   457→   457→        }
   458→   458→
   459→   459→        a.finalizeProposal(state, proposal)
   460→   460→        return nil
   461→   461→}

   463→   463→// buildUserMessage constructs the prompt for the LLM. Puts sandbox feedback

   500→   500→        }

   507→   507→                        resultStr = resultStr[:MaxFixResultChars]

   512→   512→        parts = append(parts, "## Fetched Source Code")
   513→   513→        i := 0
   514→   514→        for path, content := range state.FetchedCode {
   515→   515→                if i >= 5 {
   516→   516→                        break

   521→   521→                        content = content[:6000]

   527→   527→        parts = append(parts, "")
   528→   528→        parts = append(parts, "## Your Task")

   530→   530→        parts = append(parts, "IMPORTANT: Use the EXACT file path from '### <path>' above in your files_to_change.")
   531→   531→        parts = append(parts, "Do NOT use container paths (like /app/app/main.py) — use the repo path (like app/main.py).")
   532→   532→        return strings.Join(parts, "\n")
   533→   533→}

   535→   535→// detectLanguageHint concatenates anomaly explanations (which contain the
   536→   536→// original pod logs) and runs ParseError to detect the language. Returns ""
   537→   537→// if no language can be detected.
   538→   538→func (a *FixProposer) detectLanguageHint(state *schemas.AgentState) string {
   539→   539→        var logs string
   540→   540→        for _, anomaly := range state.Cluster.Anomalies {
   541→   541→                logs += anomaly.Explanation + "\n"
   542→   542→        }
   543→   543→        if logs == "" {
   544→   544→                return ""
   545→   545→        }
   546→   546→        parsed := ParseError(logs)
   547→   547→        if parsed == nil {
   548→   548→                return ""
   549→   549→        }
   550→   550→        if parsed.Language == "" || parsed.Language == "unknown" {
   551→   551→                return ""
   552→   552→        }
   553→   553→        switch parsed.Language {
   554→   554→        case "python":
   555→   555→                return "Python (idiomatic errors: KeyError, ValueError, TypeError, ImportError, ZeroDivisionError, AttributeError)"
   556→   556→        case "go":
   557→   557→                return "Go (idiomatic errors: panic, runtime error: index out of range, nil pointer dereference, goroutine stack)"
   558→   558→        case "java":
   559→   559→                return "Java (idiomatic errors: NullPointerException, ArrayIndexOutOfBoundsException, ClassNotFoundException, IllegalArgumentException)"
   560→   560→        case "javascript":
   561→   561→                return "JavaScript/TypeScript (idiomatic errors: TypeError, ReferenceError, SyntaxError, UnhandledPromiseRejection)"
   562→   562→        case "rust":
   563→   563→                return "Rust (idiomatic errors: panic, panicked at, unwrap on None/Err, thread 'main' panicked)"
   564→   564→        case "ruby":
   565→   565→                return "Ruby (idiomatic errors: NoMethodError, NameError, ArgumentError, TypeError)"
   566→   566→        case "c":
   567→   567→                return "C/C++ (idiomatic errors: SIGSEGV, SIGABRT, Segmentation fault, assertion failed)"
   568→   568→        default:
   569→   569→                return parsed.Language
   570→   570→        }
   571→   571→}

   573→   573→// llmCodeFix is used for roll_forward cases where we still need an
   574→   574→// LLM-generated patch in addition to the roll_forward action.
   575→   575→func (a *FixProposer) llmCodeFix(ctx context.Context, state *schemas.AgentState, proposal *schemas.FixProposal) {
   576→   576→        userMsg := a.buildUserMessage(state)
   577→   577→        result, err := a.router.FrontierJSON(ctx, fixProposerSystemPrompt, userMsg, 16000)
   578→   578→        if err != nil {
   579→   579→                proposal.Description = fmt.Sprintf("%s (LLM code generation failed — needs human)", proposal.Description)
   580→   580→                return
   581→   581→        }
   582→   582→        rawFiles, _ := result["files_to_change"].([]any)
   583→   583→        for _, rf := range rawFiles {
   584→   584→                fcMap, ok := rf.(map[string]any)
   585→   585→                if !ok {
   586→   586→                        continue
   587→   587→                }
   588→   588→                path, _ := fcMap["path"].(string)
   589→   589→                content, _ := fcMap["new_content"].(string)
   590→   590→
   591→   591→                validated := validateFilePath(path)
   592→   592→                if validated != "" {
   593→   593→                        proposal.FilesToChange = append(proposal.FilesToChange, schemas.FileChange{Path: validated, NewContent: content})
   594→   594→                }
   595→   595→        }
   596→   596→        if desc, ok := result["description"].(string); ok && desc != "" {
   597→   597→                proposal.Description = desc
   598→   598→        }
   599→   599→}

   692→   692→// extractMetadataJSON extracts the small JSON metadata object from the LLM
   693→   693→// output. The LLM outputs {"fix_type":"code_fix",...} before the SEARCH/REPLACE
   694→   694→// blocks. If the output starts with SEARCH/REPLACE blocks (no JSON), returns
   695→   695→// an empty map.
   696→   696→func extractMetadataJSON(rawOutput string) map[string]any {
   697→   697→        if strings.Contains(rawOutput, "<<<<<<< SEARCH") {
   698→   698→                idx := strings.Index(rawOutput, "<<<<<<< SEARCH")
   699→   699→                if idx > 0 {
   700→   700→                        beforeBlocks := rawOutput[:idx]
   701→   701→                        return llm.ExtractJSON(beforeBlocks)
   702→   702→                }
   703→   703→        }
   704→   704→        return llm.ExtractJSON(rawOutput)
   705→   705→}

   707→   707→// normalizeFilePath matches the LLM-provided file path against the keys in
   708→   708→// FetchedCode. LLMs often use container paths (e.g. /app/app/main.py) or
   709→   709→// add/remove leading directories. Returns the matching fetched-code key, or
   710→   710→// the original path if no match.
   711→   711→func normalizeFilePath(path string, fetchedCode map[string]string) string {
   712→   712→        if path == "" {
   713→   713→                return ""
   714→   714→        }
   715→   715→        // Exact match.
   716→   716→        if _, ok := fetchedCode[path]; ok {
   717→   717→                return path
   718→   718→        }
   719→   719→        // Path is a suffix of a fetched key (LLM used /app/app/main.py, key is app/main.py).
   720→   720→        for key := range fetchedCode {
   721→   721→                if strings.HasSuffix(key, path) || strings.HasSuffix(path, key) {
   722→   722→                        return key
   723→   723→                }
   724→   724→        }
   725→   725→        // Strip leading directories from path.
   726→   726→        parts := strings.Split(path, "/")
   727→   727→        for i := 0; i < len(parts); i++ {
   728→   728→                shortPath := strings.Join(parts[i:], "/")
   729→   729→                if _, ok := fetchedCode[shortPath]; ok {
   730→   730→                        return shortPath
   731→   731→                }
   732→   732→        }
   733→   733→        // No match — return as-is (the apply step will fail with a clear error).
   734→   734→        return path
   735→   735→}

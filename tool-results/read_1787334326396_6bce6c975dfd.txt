     1→     1→// Package main — wiring constructor.
     2→     2→//
     3→     3→// wire.go contains the wire() function that instantiates every Sentinel
     4→     4→// component in dependency order and returns a fully-constructed *wiring.
     5→     5→// Split from main.go so main.go stays a small, readable entry point
     6→     6→// (main + run); the construction logic is linear and reads top-to-bottom
     7→     7→// here.
     8→     8→//
     9→     9→// Construction is in five phases:
    10→    10→//
    11→    11→//  1. Storage + infra (store, k8s, router, embedder, vectorStore, pipeline).
    12→    12→//  2. Executors + safety (ghExec, k8sExec, sandbox, kaniko, policy).
    13→    13→//  3. Detection + correlation (tracker, clusterer, detector, correlator,
    14→    14→//     outcomeWatcher, alertmanager receiver, CRD watcher).
    15→    15→//  4. Tier 1/2 features (attest, prometheus, postmerge, slo, causal,
    16→    16→//     admission, llmcost-implicit, chaos).
    17→    17→//  5. API surface (prreview, chatops, automerge, slackbot, apiServer).
    18→    18→//
    19→    19→// The runtimecfg apply callback is registered LAST, after every component
    20→    20→// it mutates has been constructed.

    66→    66→func wire(cfg *config.Config, listenAddr string) (*wiring, error) {
    67→    67→        w := &wiring{
    68→    68→                cfg:         cfg,
    69→    69→                broadcaster: events.Noop(),
    70→    70→                ready:       &atomic.Bool{},
    71→    71→        }
    72→    72→
    73→    73→        // --- Phase 1: Storage + infra ---
    74→    74→
    75→    75→        store, err := feedback.New(cfg.SQLitePath)
    76→    76→        if err != nil {
    77→    77→                return nil, fmt.Errorf("storage init: %w", err)
    78→    78→        }
    79→    79→        w.store = store
    80→    80→
    81→    81→        k8sClient, err := k8s.New()
    82→    82→        if err != nil {
    83→    83→                _ = store.Close()
    84→    84→                return nil, fmt.Errorf("k8s client: %w", err)
    85→    85→        }
    86→    86→        w.k8sClient = k8sClient
    87→    87→
    88→    88→        router := llm.NewWithProvider(cfg.LLMProvider, cfg.ResolvedBaseURL(), cfg.LLMAPIKey, cfg.LLMFastModel, cfg.LLMFrontierModel)
    89→    89→        router.SetRetryConfig(cfg.LLMMaxRetries, time.Duration(cfg.LLMRetryBaseDelayMs)*time.Millisecond)
    90→    90→        if cfg.LLMProvider == "bedrock" {
    91→    91→                router.SetBedrockRegion(func() string {
    92→    92→                        r := os.Getenv("AWS_REGION")
    93→    93→                        if r == "" {
    94→    94→                                return "us-east-1"
    95→    95→                        }
    96→    96→                        return r
    97→    97→                }())
    98→    98→        }
    99→    99→        if len(cfg.LLMProviders) > 0 {
   100→   100→                pool := llm.NewPool(cfg.LLMProviders)
   101→   101→                pool.SetRetryConfig(cfg.LLMMaxRetries, time.Duration(cfg.LLMRetryBaseDelayMs)*time.Millisecond)
   102→   102→                router.SetPool(pool)
   103→   103→                logging.Info("LLM failover pool enabled",
   104→   104→                        "providers", len(cfg.LLMProviders),
   105→   105→                        "mode", "multi-provider",
   106→   106→                )
   107→   107→        } else {
   108→   108→                logging.Info("LLM failover pool disabled (single-provider mode)",
   109→   109→                        "provider", cfg.LLMProvider,
   110→   110→                        "hint", "set SENTINEL_LLM_PROVIDERS to enable multi-key failover",
   111→   111→                )
   112→   112→        }
   113→   113→        w.router = router
   114→   114→
   115→   115→        llm.SetMetricsRecorder(func(model, status string, duration time.Duration, inputTokens, outputTokens int) {
   116→   116→                metrics.RecordLLMCall(model, status, duration)
   117→   117→                if status == "success" && (inputTokens > 0 || outputTokens > 0) {
   118→   118→                        metrics.RecordLLMTokens(model, inputTokens, outputTokens)
   119→   119→                }
   120→   120→        })
   121→   121→        llm.SetInFlightHooks(metrics.LLMCallStart, metrics.LLMCallEnd)
   122→   122→
   123→   123→        // Embedder + vector store. Auto-detects remote embeddings when an API
   124→   124→        // key is set; falls back to TF-IDF otherwise.
   125→   125→        embAPIKey := cfg.EmbeddingAPIKey
   126→   126→        if embAPIKey == "" && cfg.LLMEmbeddingModel != "" {
   127→   127→                embAPIKey = cfg.LLMAPIKey
   128→   128→        }
   129→   129→        embBaseURL := cfg.EmbeddingBaseURL
   130→   130→        if embBaseURL == "" && cfg.LLMEmbeddingModel != "" {
   131→   131→                embBaseURL = cfg.ResolvedBaseURL()
   132→   132→        }
   133→   133→        embedder, err := emb.New(emb.Config{
   134→   134→                Provider:  cfg.EmbeddingProvider,
   135→   135→                Model:     cfg.LLMEmbeddingModel,
   136→   136→                APIKey:    embAPIKey,
   137→   137→                BaseURL:   embBaseURL,
   138→   138→                Dimension: cfg.LLMEmbeddingDim,
   139→   139→        })
   140→   140→        if err != nil {
   141→   141→                logging.Error("failed to create embedder, using TF-IDF fallback", "err", err)
   142→   142→                embedder, _ = emb.New(emb.Config{Provider: "tfidf", Dimension: cfg.LLMEmbeddingDim})
   143→   143→        }
   144→   144→        logging.Info("embedding pipeline configured", "provider", embedder.Name(), "semantic", embedder.IsSemantic())
   145→   145→        w.embedder = embedder
   146→   146→
   147→   147→        vectorStore := feedback.NewVectorStore(embedder)
   148→   148→        w.vectorStore = vectorStore
   149→   149→
   150→   150→        if cfg.LLMSemanticCacheEnabled && embedder.IsSemantic() {
   151→   151→                cache := llm.NewSemanticCache(embedder, float32(cfg.LLMSemanticCacheThreshold), 24*time.Hour, 1000)
   152→   152→                router.SetCache(cache)
   153→   153→                logging.Info("LLM semantic cache enabled", "threshold", cfg.LLMSemanticCacheThreshold)
   154→   154→        } else if cfg.LLMSemanticCacheEnabled && !embedder.IsSemantic() {
   155→   155→                logging.Warn("LLM semantic cache requested but embeddings are non-semantic (TF-IDF) — cache disabled. Set SENTINEL_LLM_EMBEDDING_MODEL to enable.")
   156→   156→        }
   157→   157→
   158→   158→        // Pipeline + safety policy.
   159→   159→        // agentsVS is the agents.VectorStore adapter over vectorStore. Held
   160→   160→        // on the wiring struct so applyRuntimeConfig can reuse it when
   161→   161→        // hot-toggling the agent loop.
   162→   162→        agentsVS := &agentsVectorStore{vs: vectorStore}
   163→   163→        w.agentsVS = agentsVS
   164→   164→        pipeline := agents.NewPipeline(router, k8sClient, cfg.GitHubToken, store, agentsVS)
   165→   165→        pipeline.SetConfig(
   166→   166→                cfg.InvestigatorTimeoutSec,
   167→   167→                cfg.EvidenceGathererTimeoutSec,
   168→   168→                cfg.AnalystTimeoutSec,
   169→   169→                cfg.FixProposerTimeoutSec,
   170→   170→                cfg.ConfidenceLoopBackThreshold,
   171→   171→        )
   172→   172→        if cfg.PrometheusURL != "" {
   173→   173→                pipeline.SetPrometheusURL(cfg.PrometheusURL)
   174→   174→        }
   175→   175→        pipeline.SetAllowedRepos(cfg.AllowedGitHubRepos)
   176→   176→        pipeline.SetGitHost(cfg.GitHost)
   177→   177→        pipeline.SetDefaultBranch(cfg.DefaultBranch)
   178→   178→        pipeline.SetLLMPriorWeight(cfg.LLMPriorWeight)
   179→   179→        pipeline.SetMaxIterations(cfg.MaxPipelineIterations)
   180→   180→
   181→   181→        // Phase 11: when SENTINEL_USE_AGENT_LOOP=true, swap the pipeline's
   182→   182→        // hypothesis/root-cause phase to the N-turn agent loop. The adapter
   183→   183→        // reuses the same router, K8s client, GitHub token, and vector
   184→   184→        // store as the legacy investigator so there's no duplicate clone
   185→   185→        // rate-limit pressure. Toggable at runtime via /settings
   186→   186→        // (use_agent_loop) — see applyRuntimeConfig in callbacks.go.
   187→   187→        if cfg.UseAgentLoop {
   188→   188→                agentAdapter := agent.NewPipelineAdapter(
   189→   189→                        router, k8sClient, cfg.GitHubToken,
   190→   190→                        agentsVS,
   191→   191→                        agent.AgentConfig{
   192→   192→                                MaxTurns:         agent.DefaultMaxTurns,
   193→   193→                                TurnTimeout:      time.Duration(cfg.InvestigatorTimeoutSec) * time.Second,
   194→   194→                                CompactThreshold: agent.DefaultCompactThreshold,
   195→   195→                        },
   196→   196→                )
   197→   197→                agentAdapter.SetAllowedRepos(cfg.AllowedGitHubRepos)
   198→   198→                agentAdapter.SetGitHost(cfg.GitHost)
   199→   199→                agentAdapter.SetDefaultBranch(cfg.DefaultBranch)
   200→   200→                pipeline.SetAgentLoopRunner(agentAdapter)
   201→   201→                logging.Info("agent loop enabled (Phase 11) — pipeline will use N-turn investigator",
   202→   202→                        "max_turns", agent.DefaultMaxTurns,
   203→   203→                        "turn_timeout_sec", cfg.InvestigatorTimeoutSec)
   204→   204→        }
   205→   205→        w.pipeline = pipeline
   206→   206→
   207→   207→        policy := safety.NewPolicy(cfg.ProductionNamespaces, cfg.ConfidenceAutonomous, cfg.ConfidenceHumanReview).
   208→   208→                WithHighRiskServices(cfg.HighRiskServices)
   209→   209→        w.policy = policy
   210→   210→
   211→   211→        // L2 safe-actions controller.
   212→   212→        w.safeActions = safeactions.New(k8sClient.Clientset(), safeactions.Config{
   213→   213→                Enabled:           cfg.SafeActionsEnabled,
   214→   214→                AllowedNamespaces: cfg.SafeActionsNamespaces,
   215→   215→                MaxActionsPerHour: cfg.SafeActionsMaxPerHour,
   216→   216→                MaxScaleDeltaPct:  cfg.SafeActionsMaxScaleDeltaPct,
   217→   217→        })
   218→   218→        if cfg.SafeActionsEnabled {
   219→   219→                logging.Info("L2 safe-actions enabled",
   220→   220→                        "namespaces", cfg.SafeActionsNamespaces,
   221→   221→                        "max_per_hour", cfg.SafeActionsMaxPerHour,
   222→   222→                )
   223→   223→        }
   224→   224→
   225→   225→        // --- Phase 2: Executors ---
   226→   226→
   227→   227→        ghExec := executors.NewGitHub(cfg.GitHubToken).
   228→   228→                WithAPIURL(cfg.GitHubAPIURL).
   229→   229→                WithGitHost(cfg.GitHost).
   230→   230→                WithDefaultBranch(cfg.DefaultBranch)
   231→   231→        w.ghExec = ghExec
   232→   232→
   233→   233→        w.k8sExec = executors.NewK8s(k8sClient.Clientset())
   234→   234→
   235→   235→        w.sandbox = executors.NewSandbox(k8sClient.Clientset(), cfg.SandboxNamespacePrefix, cfg.EnableTrafficMirroring).
   236→   236→                WithProbePaths(cfg.SandboxProbePaths).
   237→   237→                WithSentinelNamespace(cfg.Namespace)
   238→   238→
   239→   239→        w.kaniko = executors.NewKaniko(k8sClient.Clientset(), cfg.Namespace).
   240→   240→                WithDockerHubUsername(cfg.DockerHubUsername).
   241→   241→                WithRegistry(cfg.KanikoRegistry).
   242→   242→                WithInitImage(cfg.KanikoInitImage).
   243→   243→                WithExecutorImage(cfg.KanikoExecutorImage).
   244→   244→                WithGitHost(cfg.GitHost).
   245→   245→                WithGitHubToken(cfg.GitHubToken)
   246→   246→
   247→   247→        if cfg.AutoRemediationEnabled {
   248→   248→                w.remediation = executors.NewRemediationController(k8sClient.Clientset(), cfg.SkipNamespaces)
   249→   249→        }
   250→   250→
   251→   251→        // --- Phase 3: Detection + correlation ---
   252→   252→
   253→   253→        w.tracker = detection.NewActiveTracker()
   254→   254→        w.alertSender = agents.NewAlertSender(cfg.SlackWebhook)
   255→   255→
   256→   256→        w.outcomeWatcher = feedback.NewOutcomeWatcher(k8sClient.Clientset(), store)
   257→   257→        if cfg.OutcomeWatchDurationMin > 0 || cfg.OutcomeCheckIntervalSec > 0 {
   258→   258→                w.outcomeWatcher.SetDurations(
   259→   259→                        time.Duration(cfg.OutcomeWatchDurationMin)*time.Minute,
   260→   260→                        time.Duration(cfg.OutcomeCheckIntervalSec)*time.Second,
   261→   261→                        0, // Phase 2 long watch — keep default 24h
   262→   262→                        0, // Phase 2 long interval — keep default 5min
   263→   263→                )
   264→   264→        }
   265→   265→
   266→   266→        // Load existing vectors into the HNSW index in the background. Runs
   267→   267→        // after a 2s delay so the SQLite pool is warm.
   268→   268→        go func() {
   269→   269→                defer recoverPanic("vector store loader")
   270→   270→                time.Sleep(2 * time.Second)
   271→   271→                loadCtx, loadCancel := context.WithTimeout(context.Background(), 60*time.Second)
   272→   272→                defer loadCancel()
   273→   273→                if err := vectorStore.LoadFromDB(loadCtx, store); err != nil {
   274→   274→                        logging.Error("vector store load failed", "error", err)
   275→   275→                }
   276→   276→        }()
   277→   277→
   278→   278→        maxConcurrent := cfg.MaxConcurrentInvestigations
   279→   279→        if maxConcurrent <= 0 {
   280→   280→                maxConcurrent = 3
   281→   281→        }
   282→   282→        w.invSem = make(chan struct{}, maxConcurrent)
   283→   283→
   284→   284→        // Correlator must be constructed before the clusterer handler because
   285→   285→        // the handler captures it. pkgCausalAnalyzer is set later in Phase 4
   286→   286→        // (Tier 1); early cascades just skip causal attribution.
   287→   287→        w.correlator = correlation.New(func(event correlation.CorrelationEvent) {
   288→   288→                logging.Warn("cascade event detected",
   289→   289→                        "services", event.Services,
   290→   290→                        "likely_cause", event.LikelyCause,
   291→   291→                        "incident_count", event.IncidentCount,
   292→   292→                )
   293→   293→                w.alertSender.SendAlert(
   294→   294→                        event.LikelyCause,
   295→   295→                        "cascade",
   296→   296→                        fmt.Sprintf("Cascade detected: %d services failing. Likely cause: %s", event.IncidentCount, event.LikelyCause),
   297→   297→                        "",
   298→   298→                )
   299→   299→                if w.causalAnalyzer != nil {
   300→   300→                        cascade := causal.Cascade{
   301→   301→                                ID:        uuid.New().String(),
   302→   302→                                StartedAt: event.Timestamp,
   303→   303→                        }
   304→   304→                        for _, svc := range event.Services {
   305→   305→                                ns := "default"
   306→   306→                                for _, n := range event.Namespaces {
   307→   307→                                        ns = n
   308→   308→                                        break
   309→   309→                                }
   310→   310→                                cascade.Anomalies = append(cascade.Anomalies, causal.Anomaly{
   311→   311→                                        Node:      causal.Node{Service: svc, Namespace: ns},
   312→   312→                                        Metric:    "error_rate",
   313→   313→                                        StartedAt: event.Timestamp,
   314→   314→                                        Magnitude: 1.0,
   315→   315→                                })
   316→   316→                        }
   317→   317→                        if rc, err := w.causalAnalyzer.Analyze(context.Background(), cascade); err != nil {
   318→   318→                                logging.Warn("causal: analysis failed", "error", err)
   319→   319→                        } else {
   320→   320→                                logging.Info("causal: root cause attributed",
   321→   321→                                        "node", rc.Node.Service+"/"+rc.Node.Namespace,
   322→   322→                                        "confidence", rc.Confidence,
   323→   323→                                        "reasoning", rc.Reasoning,
   324→   324→                                )
   325→   325→                                if w.attestor != nil {
   326→   326→                                        _ = w.attestor.Sign(context.Background(), attest.Attestation{
   327→   327→                                                IncidentID: cascade.ID,
   328→   328→                                                Action:     "causal.root_cause_attributed",
   329→   329→                                                Reason:     rc.Reasoning,
   330→   330→                                                Outcome:    true,
   331→   331→                                                Timestamp:  time.Now(),
   332→   332→                                                Evidence: []attest.Evidence{{
   333→   333→                                                        Kind: "causal",
   334→   334→                                                        Payload: map[string]any{
   335→   335→                                                                "root_cause":   rc.Node.Service + "/" + rc.Node.Namespace,
   336→   336→                                                                "confidence":   rc.Confidence,
   337→   337→                                                                "alternatives": rc.AlternativeCandidates,
   338→   338→                                                        },
   339→   339→                                                }},
   340→   340→                                        })
   341→   341→                                }
   342→   342→                        }
   343→   343→                }
   344→   344→        })
   345→   345→
   346→   346→        // Clusterer handler: groups signals, dedupes via the tracker, and
   347→   347→        // invokes the investigation orchestrator.
   348→   348→        clustererHandler := func(service, namespace, reason, podLogs string) {
   349→   349→                annotations, _ := k8sClient.LookupWorkloadAnnotations(context.Background(), service, namespace)
   350→   350→                if annotations == nil {
   351→   351→                        annotations = map[string]string{}
   352→   352→                }
   353→   353→
   354→   354→                // When a cascade is detected (3+ services failing within the
   355→   355→                // correlator window), the correlator emits an event + we skip
   356→   356→                // the individual investigation — the cascade handler will
   357→   357→                // trigger a rollback instead.
   358→   358→                if w.correlator.RegisterIncident(service, namespace, time.Now()) {
   359→   359→                        logging.Info("cascade detected — skipping individual investigation",
   360→   360→                                "service", service,
   361→   361→                                "namespace", namespace,
   362→   362→                        )
   363→   363→                        return
   364→   364→                }
   365→   365→
   366→   366→                explanation := fmt.Sprintf("Auto-detected incident: %s", reason)
   367→   367→                if podLogs != "" {
   368→   368→                        explanation = fmt.Sprintf("%s\n\n## Pod Logs (preprocessed)\n%s", explanation, podLogs)
   369→   369→                }
   370→   370→
   371→   371→                cluster := schemas.AnomalyCluster{
   372→   372→                        ClusterID: uuid.New(),
   373→   373→                        Service:   service,
   374→   374→                        Namespace: namespace,
   375→   375→                        Anomalies: []schemas.Anomaly{
   376→   376→                                {
   377→   377→                                        AnomalyID:   uuid.New(),
   378→   378→                                        SignalID:    uuid.New(),
   379→   379→                                        Service:     service,
   380→   380→                                        Namespace:   namespace,
   381→   381→                                        Detector:    "auto_detection",
   382→   382→                                        Score:       0.9,
   383→   383→                                        Dimensions:  map[string]float64{"auto": 1.0},
   384→   384→                                        Explanation: explanation,
   385→   385→                                        Annotations: annotations,
   386→   386→                                },
   387→   387→                        },
   388→   388→                        Summary:     reason,
   389→   389→                        Severity:    schemas.SeverityWarning,
   390→   390→                        Annotations: annotations,
   391→   391→                }
   392→   392→
   393→   393→                w.alertSender.SendAlert(service, namespace, reason, "")
   394→   394→                go w.handleInvestigation(context.Background(), cluster, "auto-detected", namespace)
   395→   395→        }
   396→   396→
   397→   397→        w.clusterer = detection.NewClusterer(clustererHandler, w.tracker)
   398→   398→
   399→   399→        detectionHandler := func(service, namespace, reason, _ string) {
   400→   400→                logging.Info("detection signal",
   401→   401→                        "namespace", namespace,
   402→   402→                        "service", service,
   403→   403→                        "reason", reason,
   404→   404→                )
   405→   405→                annotations, _ := k8sClient.LookupWorkloadAnnotations(context.Background(), service, namespace)
   406→   406→                repo := ""
   407→   407→                if annotations != nil {
   408→   408→                        repo = annotations["sentinel.io/repo"]
   409→   409→                }
   410→   410→                w.clusterer.Send(context.Background(), detection.Signal{
   411→   411→                        Type:      detection.SignalLogError,
   412→   412→                        Service:   service,
   413→   413→                        Namespace: namespace,
   414→   414→                        Reason:    reason,
   415→   415→                        Repo:      repo,
   416→   416→                        Timestamp: time.Now(),
   417→   417→                })
   418→   418→        }
   419→   419→
   420→   420→        w.detector = detection.NewWithConfig(
   421→   421→                k8sClient.Clientset(), detectionHandler, cfg.SlackWebhook,
   422→   422→                time.Duration(cfg.DetectionCooldownSec)*time.Second,
   423→   423→                cfg.DetectionTailLines,
   424→   424→                cfg.SkipNamespaces,
   425→   425→        )
   426→   426→
   427→   427→        // Alertmanager webhook receiver — converts firing alerts into

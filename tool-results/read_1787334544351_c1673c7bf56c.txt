    49→// handleMetricsSummary returns a count of incidents by lifecycle bucket
    50→// (active / resolved / failed) over the most recent 100 incidents.
    51→func (s *Server) handleMetricsSummary(w http.ResponseWriter, r *http.Request) {
    52→        incidents, _ := s.store.ListRecentIncidents(r.Context(), 100)
    53→        active, resolved, failed := 0, 0, 0
    54→        for _, inc := range incidents {
    55→                status := string(inc.Status)
    56→                switch {
    57→                case strings.Contains("investigating root_caused fix_proposed approved executing", status):
    58→                        active++
    59→                case strings.Contains("verified resolved", status):
    60→                        resolved++
    61→                case strings.Contains("failed rejected escalated", status):
    62→                        failed++
    63→                }
    64→        }
    65→        writeJSON(w, 200, map[string]int{
    66→                "active":       active,
    67→                "resolved_24h": resolved,
    68→                "failed":       failed,
    69→                "total":        len(incidents),
    70→        })
    71→}

    73→// handleIncidents lists recent incidents (GET) with a limit query param.
    74→func (s *Server) handleIncidents(w http.ResponseWriter, r *http.Request) {
    75→        limit := 50
    76→        if l := r.URL.Query().Get("limit"); l != "" {
    77→                if n, err := fmt.Sscanf(l, "%d", &limit); err != nil || n == 0 {
    78→                        limit = 50
    79→                }
    80→        }
    81→        incidents, err := s.store.ListRecentIncidents(r.Context(), limit)
    82→        if err != nil {
    83→                writeJSON(w, 500, map[string]string{"error": err.Error()})
    84→                return
    85→        }
    86→        // Mirror pr_number/pr_url/etc from state.execution_result → top-level
    87→        // fields so the frontend's PRs tab + incident detail can read them
    88→        // without digging into the nested state object.
    89→        for _, inc := range incidents {
    90→                enrichIncidentPRFields(inc)
    91→        }
    92→        writeJSON(w, 200, incidents)
    93→}

    95→// handleIncident returns a single incident by full UUID or short prefix.
    96→func (s *Server) handleIncident(w http.ResponseWriter, r *http.Request) {
    97→        idStr := strings.TrimPrefix(r.URL.Path, "/incidents/")
    98→
    99→        id, err := uuid.Parse(idStr)
   100→        if err == nil {
   101→                // Full UUID — direct lookup.
   102→                incident, err := s.store.GetIncident(r.Context(), id)
   103→                if err != nil {
   104→                        writeJSON(w, 500, map[string]string{"error": err.Error()})
   105→                        return
   106→                }
   107→                if incident == nil {
   108→                        writeJSON(w, 404, map[string]string{"error": "incident not found"})
   109→                        return
   110→                }
   111→                enrichIncidentPRFields(incident)
   112→                writeJSON(w, 200, incident)
   113→                return
   114→        }
   115→
   116→        // Not a full UUID — try prefix matching. FindIncidentByPrefix requires
   117→        // ≥4 chars to prevent accidental collisions on short prefixes.
   118→        incident, err := s.store.FindIncidentByPrefix(r.Context(), idStr)
   119→        if err != nil {
   120→                writeJSON(w, 500, map[string]string{"error": err.Error()})
   121→                return
   122→        }
   123→        if incident == nil {
   124→                writeJSON(w, 404, map[string]string{"error": "no incident matches that ID prefix (need at least 4 chars)"})
   125→                return
   126→        }
   127→        enrichIncidentPRFields(incident)
   128→        writeJSON(w, 200, incident)
   129→}

   131→// enrichIncidentPRFields mirrors pr_number/pr_url/pr_branch +
   132→// sandbox_summary + sandbox_verified from the nested state.execution_result
   133→// (which is what the agents write to) onto the top-level Incident fields
   134→// (which is what the frontend reads). Keeps the JSON contract stable
   135→// without rewriting every SELECT.
   136→func enrichIncidentPRFields(inc *schemas.Incident) {
   137→        if inc == nil {
   138→                return
   139→        }
   140→        if er := inc.State.ExecutionResult; er != nil {
   141→                if inc.PRNumber == nil && er.PRNumber != nil {
   142→                        inc.PRNumber = er.PRNumber
   143→                }
   144→                if inc.PRURL == "" && er.PRURL != nil {
   145→                        inc.PRURL = *er.PRURL
   146→                }
   147→                if inc.PRBranch == "" && er.PRBranch != nil {
   148→                        inc.PRBranch = *er.PRBranch
   149→                }
   150→                if inc.SandboxSummary == "" && er.SandboxSummary != nil {
   151→                        inc.SandboxSummary = *er.SandboxSummary
   152→                }
   153→                if !inc.SandboxVerified && er.SandboxVerified {
   154→                        inc.SandboxVerified = er.SandboxVerified
   155→                }
   156→        }
   157→        if fp := inc.State.FixProposal; fp != nil {
   158→                if inc.FixType == "" {
   159→                        inc.FixType = string(fp.FixType)
   160→                }
   161→        }
   162→}

   202→// handleInvestigate is the manual-trigger endpoint. POST with a JSON body
   203→// {service, namespace, reason} triggers a new investigation.
   204→func (s *Server) handleInvestigate(w http.ResponseWriter, r *http.Request) {
   205→        if r.Method != http.MethodPost {
   206→                writeJSON(w, 405, map[string]string{"error": "method not allowed"})
   207→                return
   208→        }
   209→        var spec struct {
   210→                Service   string `json:"service"`
   211→                Namespace string `json:"namespace"`
   212→                Reason    string `json:"reason"`
   213→                Repo      string `json:"repo"`
   214→                PodLogs   string `json:"pod_logs"`
   215→        }
   216→        r.Body = http.MaxBytesReader(nil, r.Body, maxRequestBodyBytes)
   217→        if err := json.NewDecoder(r.Body).Decode(&spec); err != nil {
   218→                writeJSON(w, 400, map[string]string{"error": "invalid JSON"})
   219→                return
   220→        }
   221→        if spec.Service == "" {
   222→                writeJSON(w, 400, map[string]string{"error": "service is required"})
   223→                return
   224→        }
   225→        if spec.Namespace == "" {
   226→                spec.Namespace = "default"
   227→        }
   228→        if spec.Reason == "" {
   229→                spec.Reason = "manual trigger"
   230→        }
   231→        if s.investigateWithLogs != nil {
   232→                clusterID, err := s.investigateWithLogs(spec.Service, spec.Namespace, spec.Reason, spec.Repo, spec.PodLogs)
   233→                if err != nil {
   234→                        writeJSON(w, 500, map[string]string{"error": err.Error()})
   235→                        return
   236→                }
   237→                writeJSON(w, 200, map[string]string{"status": "triggered", "service": spec.Service, "cluster_id": clusterID})
   238→                return
   239→        }
   240→        clusterID, err := s.investigate(spec.Service, spec.Namespace, spec.Reason)
   241→        if err != nil {
   242→                writeJSON(w, 500, map[string]string{"error": err.Error()})
   243→                return
   244→        }
   245→        writeJSON(w, 200, map[string]string{
   246→                "status":     "triggered",
   247→                "service":    spec.Service,
   248→                "cluster_id": clusterID,
   249→        })
   250→}

   252→// handlePause pauses autonomous mode for a configurable duration (default 30min).
   253→func (s *Server) handlePause(w http.ResponseWriter, r *http.Request) {
   254→        if r.Method != http.MethodPost {
   255→                writeJSON(w, 405, map[string]string{"error": "method not allowed"})
   256→                return
   257→        }
   258→        if s.pauseController == nil {
   259→                writeJSON(w, 503, map[string]string{"error": "pause controller not configured"})
   260→                return
   261→        }
   262→        var req struct {
   263→                DurationMin int    `json:"duration_min"`
   264→                Reason      string `json:"reason"`
   265→        }
   266→        if r.Body != nil {
   267→                r.Body = http.MaxBytesReader(nil, r.Body, maxRequestBodyBytes)
   268→                _ = json.NewDecoder(r.Body).Decode(&req)
   269→        }
   270→        if req.DurationMin <= 0 {
   271→                req.DurationMin = 30
   272→        }
   273→        if req.Reason == "" {
   274→                req.Reason = "paused via CLI"
   275→        }
   276→        s.pauseController.Pause(time.Duration(req.DurationMin)*time.Minute, req.Reason)
   277→        writeJSON(w, 200, map[string]any{
   278→                "status":       "paused",
   279→                "duration_min": req.DurationMin,
   280→                "reason":       req.Reason,
   281→        })
   282→}

   298→// maxSSESubscribers caps concurrent SSE connections to prevent memory
   299→// exhaustion DoS. SSE is unauthenticated (when apiToken is empty) and
   300→// each subscriber pins a goroutine + 16-entry buffered channel + map
   301→// entry (~10KB). 100 subscribers is generous for a single-pod dashboard.
   302→const maxSSESubscribers = 100
   303→
   304→// handleSSE streams events to the client. Subscribes to the broadcaster,
   305→// writes "data: {...}\n\n" frames, and flushes on each event. A 15s
   306→// keep-alive ping prevents intermediate proxies from closing the connection.
   307→func (s *Server) handleSSE(w http.ResponseWriter, r *http.Request) {
   308→        w.Header().Set("Content-Type", "text/event-stream")
   309→        w.Header().Set("Cache-Control", "no-cache")
   310→        w.Header().Set("Connection", "keep-alive")
   311→        w.Header().Set("Access-Control-Allow-Origin", "*")
   312→
   313→        flusher, ok := w.(http.Flusher)
   314→        if !ok {
   315→                http.Error(w, "streaming not supported", http.StatusInternalServerError)
   316→                return
   317→        }
   318→
   319→        s.eventSubsMu.Lock()
   320→        if len(s.eventSubs) >= maxSSESubscribers {
   321→                s.eventSubsMu.Unlock()
   322→                http.Error(w, "too many SSE subscribers (max "+strconv.Itoa(maxSSESubscribers)+")", http.StatusServiceUnavailable)
   323→                return
   324→        }
   325→        ch := make(chan ServerEvent, 16)
   326→        s.eventSubs[ch] = true
   327→        s.eventSubsMu.Unlock()
   328→        defer func() {
   329→                s.eventSubsMu.Lock()
   330→                delete(s.eventSubs, ch)
   331→                s.eventSubsMu.Unlock()
   332→        }()
   333→
   334→        // Send events as unnamed SSE messages (no `event:` line) so the
   335→        // frontend's EventSource.onmessage handler fires for every event.
   336→        // The event type is embedded in the JSON payload's `type` field.
   337→        fmt.Fprintf(w, "data: {\"type\":\"connected\",\"status\":\"connected\"}\n\n")
   338→        flusher.Flush()
   339→
   340→        ticker := time.NewTicker(15 * time.Second)
   341→        defer ticker.Stop()
   342→
   343→        for {
   344→                select {
   345→                case <-r.Context().Done():
   346→                        return
   347→                case event := <-ch:
   348→                        data, _ := json.Marshal(event)
   349→                        fmt.Fprintf(w, "data: %s\n\n", data)
   350→                        flusher.Flush()
   351→                case <-ticker.C:
   352→                        fmt.Fprintf(w, ": ping\n\n")
   353→                        flusher.Flush()
   354→                }
   355→        }
   356→}

   358→// handleWebhook dispatches /webhook/github to the prreview + chatops
   359→// handlers. The body is buffered once so both handlers can read it; each
   360→// handler ignores event types it doesn't care about.
   361→func (s *Server) handleWebhook(w http.ResponseWriter, r *http.Request) {
   362→        if s.webhookHandler == nil && s.chatopsHandler == nil {
   363→                writeJSON(w, 503, map[string]string{"error": "webhook handler not configured"})
   364→                return
   365→        }
   366→
   367→        bodyBytes, err := io.ReadAll(io.LimitReader(r.Body, 1<<20)) // 1MB cap.
   368→        if err != nil {
   369→                writeJSON(w, 400, map[string]string{"error": "read body failed"})
   370→                return
   371→        }
   372→        multiReader := func() io.Reader { return bytes.NewReader(bodyBytes) }
   373→
   374→        // 1. Dispatch to prreview (predictive PR review — pull_request events).
   375→        if s.webhookHandler != nil {
   376→                req := r.Clone(r.Context())
   377→                req.Body = io.NopCloser(multiReader())
   378→                s.webhookHandler.ServeHTTP(w, req)
   379→        }
   380→
   381→        // 2. Dispatch to chatops (issue_comment events for /sentinel commands).
   382→        // Short-circuit for non-issue_comment events so we don't buffer +
   383→        // parse for events chatops doesn't care about.
   384→        if s.chatopsHandler != nil {
   385→                eventType := r.Header.Get("X-GitHub-Event")
   386→                if eventType == "issue_comment" {
   387→                        req := r.Clone(r.Context())
   388→                        req.Body = io.NopCloser(multiReader())
   389→                        s.chatopsHandler.ServeHTTP(w, req)
   390→                }
   391→        }
   392→}

   413→// handleGitHubApp manages the GitHub App config.
   414→// GET    → status (configured? app_id? slug? install_id?).
   415→// POST   → store App ID + PEM + optional install ID + webhook secret.
   416→// DELETE → remove App config (fall back to PAT).
   417→func (s *Server) handleGitHubApp(w http.ResponseWriter, r *http.Request) {
   418→        if s.githubAppManager == nil {
   419→                writeJSON(w, 503, map[string]string{"error": "github app manager not configured"})
   420→                return
   421→        }
   422→
   423→        switch r.Method {
   424→        case http.MethodGet:
   425→                status := s.githubAppManager.GetGitHubAppStatus()
   426→                writeJSON(w, 200, status)
   427→
   428→        case http.MethodPost:
   429→                r.Body = http.MaxBytesReader(nil, r.Body, maxRequestBodyBytes)
   430→                var req struct {
   431→                        AppID         string `json:"app_id"`
   432→                        PrivateKey    string `json:"private_key"`
   433→                        InstallID     string `json:"install_id"`
   434→                        WebhookSecret string `json:"webhook_secret"`
   435→                }
   436→                if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   437→                        writeJSON(w, 400, map[string]string{"error": "invalid JSON: " + err.Error()})
   438→                        return
   439→                }
   440→                if req.AppID == "" || req.PrivateKey == "" {
   441→                        writeJSON(w, 400, map[string]string{"error": "app_id and private_key are required"})
   442→                        return
   443→                }
   444→                slug, err := s.githubAppManager.SetGitHubApp(req.AppID, req.PrivateKey, req.InstallID, req.WebhookSecret)
   445→                if err != nil {
   446→                        writeJSON(w, 500, map[string]string{"error": err.Error()})
   447→                        return
   448→                }
   449→                logging.Info("api: GitHub App configured", "app_id", req.AppID, "slug", slug)
   450→                writeJSON(w, 200, map[string]string{
   451→                        "status": "configured",
   452→                        "slug":   slug,
   453→                        "bot":    slug + "[bot]",
   454→                        "hint":   "If install_id was empty, run `sentinelctl app set-install <id>` after installing the App.",
   455→                })
   456→
   457→        case http.MethodDelete:
   458→                if err := s.githubAppManager.RemoveGitHubApp(); err != nil {
   459→                        writeJSON(w, 500, map[string]string{"error": err.Error()})
   460→                        return
   461→                }
   462→                logging.Info("api: GitHub App config removed")
   463→                writeJSON(w, 200, map[string]string{
   464→                        "status":  "removed",
   465→                        "message": "Falling back to PAT.",
   466→                })
   467→
   468→        default:
   469→                writeJSON(w, 405, map[string]string{"error": "method not allowed"})
   470→        }
   471→}

   534→// handleAPIMeta returns the chatops command list + available API endpoints
   535→// so the frontend doesn't have to hardcode them. This avoids the v5 bug
   536→// where the frontend drifted from the backend command registry.
   537→func (s *Server) handleAPIMeta(w http.ResponseWriter, r *http.Request) {
   538→        // chatopsCommands is the serialized form of chatops.AllCommands(). Built
   539→        // once per request from the live registry — never hardcoded.
   540→        type chatopsCommandMeta struct {
   541→                Name        string `json:"name"`
   542→                Description string `json:"description"`
   543→                Usage       string `json:"usage"`

   601→// handlePprof delegates to net/http/pprof for production profiling. The
   602→// URL path selects the pprof function — pprof.Index handles the index
   603→// page + heap/goroutine/threadcreate/block/mutex profiles.
   604→func (s *Server) handlePprof(w http.ResponseWriter, r *http.Request) {
   605→        switch r.URL.Path {
   606→        case "/debug/pprof/cmdline":
   607→                pprof.Cmdline(w, r)
   608→        case "/debug/pprof/profile":
   609→                pprof.Profile(w, r)
   610→        case "/debug/pprof/symbol":
   611→                pprof.Symbol(w, r)
   612→        case "/debug/pprof/trace":
   613→                pprof.Trace(w, r)
   614→        default:
   615→                pprof.Index(w, r)
   616→        }
   617→}

    84→    84→// AgentConfig configures the agent loop.

   125→   125→func New(router *llm.Router, cfg AgentConfig) *Agent {
   126→   126→        if cfg.MaxTurns <= 0 {
   127→   127→                cfg.MaxTurns = DefaultMaxTurns
   128→   128→        }
   129→   129→        if cfg.TurnTimeout <= 0 {
   130→   130→                cfg.TurnTimeout = DefaultTurnTimeout
   131→   131→        }
   132→   132→        if cfg.CompactThreshold <= 0 {
   133→   133→                cfg.CompactThreshold = DefaultCompactThreshold
   134→   134→        }
   135→   135→        if cfg.NumPlans <= 0 {
   136→   136→                cfg.NumPlans = 1 // v2 default: single plan
   137→   137→        }
   138→   138→        if cfg.CriticThreshold <= 0 {
   139→   139→                cfg.CriticThreshold = 0.0 // v2 default: critic never runs
   140→   140→        }
   141→   141→        return &Agent{
   142→   142→                router: router,
   143→   143→                cfg:    cfg,
   144→   144→        }
   145→   145→}

   160→   160→// Run executes the agent loop with planning, stuck detection, and retry.
   161→   161→func (a *Agent) Run(ctx context.Context, systemPrompt string) (*InvestigationResult, error) {
   162→   162→        var turns []Turn
   163→   163→        var conversation []map[string]string
   164→   164→
   165→   165→        conversation = append(conversation, map[string]string{
   166→   166→                "role":    "system",
   167→   167→                "content": systemPrompt + "\n\n" + agentSystemRole,
   168→   168→        })
   169→   169→
   170→   170→        toolsJSON, _ := json.Marshal(a.tools)
   171→   171→        conversation = append(conversation, map[string]string{
   172→   172→                "role":    "user",
   173→   173→                "content": fmt.Sprintf("Available tools:\n%s\n\n%s\n\nStart investigating.", toolsJSON, agentInstructions),
   174→   174→        })
   175→   175→
   176→   176→        // Phase 1: Planning.
   177→   177→        // When NumPlans=1 (default, v2 behavior): generate a single plan.
   178→   178→        // When NumPlans>1 (opt-in, for large models): generate N plans,
   179→   179→        // pick the best by specificity scoring (simplified MCTS).
   180→   180→        podLogs := ""
   181→   181→        for _, msg := range conversation {
   182→   182→                if msg["role"] == "system" {
   183→   183→                        podLogs = msg["content"]
   184→   184→                        break
   185→   185→                }
   186→   186→        }
   187→   187→
   188→   188→        var plan string
   189→   189→        if a.cfg.NumPlans > 1 {
   190→   190→                plans := a.generatePlans(ctx, conversation, a.cfg.NumPlans)
   191→   191→                if len(plans) > 0 {
   192→   192→                        plan = pickBestPlan(plans, podLogs)
   193→   193→                }
   194→   194→        } else {
   195→   195→                // Single plan — one LLM call (v2 behavior).
   196→   196→                planningCtx, planningCancel := context.WithTimeout(ctx, a.cfg.TurnTimeout)
   197→   197→                p, pErr := a.router.FrontierText(planningCtx,
   198→   198→                        "You are an autonomous SRE agent. Read the crash context and write a brief 2-3 step investigation plan. Which files will you read? What will you search for? Be specific.",
   199→   199→                        formatConversation(conversation), 500)
   200→   200→                planningCancel()
   201→   201→                if pErr != nil {
   202→   202→                        logging.Warn("agent: planning call failed, skipping plan", "err", pErr)
   203→   203→                } else {
   204→   204→                        plan = p
   205→   205→                }
   206→   206→        }
   207→   207→
   208→   208→        if plan != "" {
   209→   209→                conversation = append(conversation,
   210→   210→                        map[string]string{"role": "assistant", "content": plan},
   211→   211→                        map[string]string{"role": "user", "content": fmt.Sprintf("Here is your investigation plan:\n%s\n\nNow execute it. Call tools one at a time. Respond with JSON only.", truncate(plan, 500))})
   212→   212→                logging.Info("agent: plan generated", "plans", a.cfg.NumPlans, "plan", truncate(plan, 100))
   213→   213→        } else {
   214→   214→                logging.Warn("agent: planning failed, starting without plan")
   215→   215→        }
   216→   216→
   217→   217→        // Phase 2: Execution loop with stuck detection.
   218→   218→        stuckDetector := NewStuckDetector()
   219→   219→        criticUsed := false // only run the critic once per investigation
   220→   220→
   221→   221→        for turnNum := 1; turnNum <= a.cfg.MaxTurns; turnNum++ {
   222→   222→                // Inner retry loop for malformed LLM output.
   223→   223→                var response string
   224→   224→                var parsed llmResponse
   225→   225→                var parseOK bool
   226→   226→
   227→   227→                for retry := 0; retry <= maxRetriesPerTurn; retry++ {
   228→   228→                        turnCtx, cancel := context.WithTimeout(ctx, a.cfg.TurnTimeout)
   229→   229→                        var llmErr error
   230→   230→                        response, llmErr = a.router.FrontierText(turnCtx,
   231→   231→                                agentSystemRole, formatConversation(conversation), 2000)
   232→   232→                        cancel()
   233→   233→
   234→   234→                        if llmErr != nil {
   235→   235→                                logging.Error("agent: LLM call failed", "turn", turnNum, "retry", retry, "err", llmErr)
   236→   236→                                break
   237→   237→                        }
   238→   238→
   239→   239→                        parsed, parseOK = parseLLMResponse(response)
   240→   240→                        if parseOK {
   241→   241→                                break
   242→   242→                        }
   243→   243→
   244→   244→                        logging.Warn("agent: malformed LLM output, retrying", "turn", turnNum, "retry", retry, "response_preview", truncate(response, 100))
   245→   245→                        conversation = append(conversation,
   246→   246→                                map[string]string{"role": "assistant", "content": response},
   247→   247→                                map[string]string{"role": "user", "content": correctionPrompt})
   248→   248→                }
   249→   249→
   250→   250→                if !parseOK {
   251→   251→                        turns = append(turns, Turn{TurnNumber: turnNum, LLMResponse: response})
   252→   252→                        logging.Warn("agent: turn produced no valid output after retries", "turn", turnNum)
   253→   253→                        continue
   254→   254→                }
   255→   255→
   256→   256→                if err := ctx.Err(); err != nil {
   257→   257→                        break
   258→   258→                }
   259→   259→
   260→   260→                turn := Turn{TurnNumber: turnNum, LLMResponse: response}
   261→   261→
   262→   262→                if parsed.Done {
   263→   263→                        turns = append(turns, turn)
   264→   264→
   265→   265→                        // Multi-agent critic: verify the root cause before
   266→   266→                        // declaring done. Only runs when CriticThreshold > 0
   267→   267→                        // AND confidence is below the threshold. Default
   268→   268→                        // (CriticThreshold=0.0) = critic never runs = v2 behavior.
   269→   269→                        // Only useful on models smart enough to evaluate root
   270→   270→                        // causes (70B+). On small models, the critic rejects
   271→   271→                        // correct answers and wastes turns.
   272→   272→                        if a.cfg.CriticThreshold > 0 && !criticUsed && parsed.Confidence < a.cfg.CriticThreshold && turnNum < a.cfg.MaxTurns {
   273→   273→                                criticUsed = true
   274→   274→                                approved, critique := a.criticReview(ctx, parsed.RootCause, parsed.FilesToChange, podLogs, conversation)
   275→   275→                                if !approved {
   276→   276→                                        logging.Warn("agent: critic rejected root cause", "turn", turnNum, "critique", critique)
   277→   277→                                        conversation = append(conversation,
   278→   278→                                                map[string]string{"role": "user", "content": fmt.Sprintf(
   279→   279→                                                        "[CRITIC] Your root cause was rejected: %s\n\nContinue investigating. Try a different approach.", critique)})
   280→   280→                                        continue
   281→   281→                                }
   282→   282→                        }
   283→   283→
   284→   284→                        criticApproved := criticUsed || parsed.Confidence >= 0.80
   285→   285→                        logging.Info("agent: investigation complete", "turns", turnNum, "root_cause", parsed.RootCause, "confidence", parsed.Confidence, "critic_approved", criticApproved)
   286→   286→                        return &InvestigationResult{
   287→   287→                                RootCause:   parsed.RootCause,
   288→   288→                                Confidence:  parsed.Confidence,
   289→   289→                                SourceFiles: parsed.FilesToChange,
   290→   290→                                Hypotheses:  []string{parsed.RootCause},
   291→   291→                                TurnsUsed:   turnNum,
   292→   292→                                Trace:       turns,
   293→   293→                        }, nil
   294→   294→                }
   295→   295→
   296→   296→                if parsed.Tool != "" {
   297→   297→                        turn.ToolCall = &ToolCall{Tool: parsed.Tool, Args: parsed.Args}
   298→   298→                        result := a.executeTool(ctx, parsed.Tool, parsed.Args)
   299→   299→                        turn.ToolResult = &result
   300→   300→
   301→   301→                        // Record for stuck detection.
   302→   302→                        stuckDetector.Record(parsed.Tool, parsed.Args, result)
   303→   303→
   304→   304→                        // Check if stuck — inject recovery prompt if needed.
   305→   305→                        stuckMsg := stuckDetector.CheckStuck()
   306→   306→                        toolResultMsg := fmt.Sprintf("Tool %s result:\n%s", parsed.Tool, result.Output)
   307→   307→                        if stuckMsg != "" {
   308→   308→                                toolResultMsg += "\n\n" + stuckMsg
   309→   309→                                logging.Warn("agent: stuck detector triggered", "turn", turnNum, "msg", truncate(stuckMsg, 80))
   310→   310→                        }
   311→   311→
   312→   312→                        conversation = append(conversation,
   313→   313→                                map[string]string{"role": "assistant", "content": response},
   314→   314→                                map[string]string{"role": "user", "content": toolResultMsg})
   315→   315→                } else {
   316→   316→                        conversation = append(conversation,
   317→   317→                                map[string]string{"role": "assistant", "content": response})
   318→   318→                }
   319→   319→
   320→   320→                turns = append(turns, turn)
   321→   321→                logging.Info("agent: turn complete", "turn", turnNum, "tool", parsed.Tool, "done", parsed.Done)
   322→   322→
   323→   323→                if turnNum >= a.cfg.CompactThreshold && turnNum%5 == 0 {
   324→   324→                        conversation = a.condenseContext(ctx, conversation, systemPrompt)
   325→   325→                }
   326→   326→        }
   327→   327→
   328→   328→        // Phase 3: Turn cap reached.
   329→   329→        logging.Warn("agent: hit turn cap without completion", "max_turns", a.cfg.MaxTurns)
   330→   330→
   331→   331→        // Guard against the edge case where the very first LLM call failed
   332→   332→        // (turns is empty). Without this, turns[len(turns)-1] panics with
   333→   333→        // "index out of range [-1]". This can happen when the LLM endpoint
   334→   334→        // is unreachable or the context is cancelled before the first
   335→   335→        // response arrives.
   336→   336→        lastResponse := "(no turns completed — first LLM call failed)"
   337→   337→        turnsUsed := a.cfg.MaxTurns
   338→   338→        if len(turns) > 0 {
   339→   339→                lastResponse = truncate(turns[len(turns)-1].LLMResponse, 200)
   340→   340→        } else {
   341→   341→                // If no turns completed, TurnsUsed should reflect that we
   342→   342→                // didn't actually run the full loop — we broke on the first
   343→   343→                // error. Report 1 so callers know the loop was entered but
   344→   344→                // immediately failed.
   345→   345→                turnsUsed = 1
   346→   346→        }
   347→   347→        return &InvestigationResult{
   348→   348→                RootCause:   fmt.Sprintf("Agent did not complete within %d turns. Last response: %s", a.cfg.MaxTurns, lastResponse),
   349→   349→                Confidence:  0.0,
   350→   350→                SourceFiles: nil,
   351→   351→                TurnsUsed:   turnsUsed,
   352→   352→                Trace:       turns,
   353→   353→        }, nil
   354→   354→}

   356→   356→func (a *Agent) executeTool(ctx context.Context, name string, args map[string]any) ToolResult {
   357→   357→        fn, ok := a.toolFuncs[name]
   358→   358→        if !ok {
   359→   359→                return ToolResult{
   360→   360→                        Tool:    name,
   361→   361→                        Success: false,
   362→   362→                        Output:  fmt.Sprintf("Unknown tool: %s", name),
   363→   363→                }
   364→   364→        }
   365→   365→        return fn(ctx, args)
   366→   366→}

   368→   368→type llmResponse struct {
   369→   369→        Done          bool           `json:"done"`
   370→   370→        Tool          string         `json:"tool"`
   371→   371→        Args          map[string]any `json:"args"`
   372→   372→        RootCause     string         `json:"root_cause"`
   373→   373→        Confidence    float64        `json:"confidence"`
   374→   374→        FilesToChange []string       `json:"files_to_change"`

   376→   376→
   377→   377→// parseLLMResponse tries to extract a valid tool call or done signal.
   378→   378→// Returns (result, ok) — ok=false means retry with correction prompt.
   379→   379→func parseLLMResponse(response string) (llmResponse, bool) {
   380→   380→        var result llmResponse
   381→   381→
   382→   382→        // Strategy 1: direct parse.
   383→   383→        if err := json.Unmarshal([]byte(response), &result); err == nil {
   384→   384→                return result, true
   385→   385→        }
   386→   386→
   387→   387→        // Strategy 2: extract from markdown code block.
   388→   388→        if code := extractCodeBlock(response); code != "" {
   389→   389→                if err := json.Unmarshal([]byte(code), &result); err == nil {
   390→   390→                        return result, true
   391→   391→                }
   392→   392→        }
   393→   393→
   394→   394→        // Strategy 3: extract first balanced JSON object.
   395→   395→        if obj := extractFirstJSONObject(response); obj != "" {
   396→   396→                if err := json.Unmarshal([]byte(obj), &result); err == nil {
   397→   397→                        return result, true
   398→   398→                }
   399→   399→        }
   400→   400→
   401→   401→        return llmResponse{}, false
   402→   402→}
   403→   403→
   404→   404→func extractCodeBlock(response string) string {
   405→   405→        startMarkers := []string{"```json\n", "```json\r\n", "```\n", "```"}
   406→   406→        for _, marker := range startMarkers {
   407→   407→                if idx := strings.Index(response, marker); idx >= 0 {
   408→   408→                        rest := response[idx+len(marker):]
   409→   409→                        if end := strings.Index(rest, "```"); end >= 0 {
   410→   410→                                return strings.TrimSpace(rest[:end])
   411→   411→                        }
   412→   412→                }
   413→   413→        }
   414→   414→        return ""
   415→   415→}
   416→   416→
   417→   417→func extractFirstJSONObject(response string) string {
   418→   418→        start := strings.Index(response, "{")
   419→   419→        if start < 0 {
   420→   420→                return ""
   421→   421→        }
   422→   422→        depth := 0
   423→   423→        inString := false
   424→   424→        escaped := false
   425→   425→        for i := start; i < len(response); i++ {
   426→   426→                ch := response[i]
   427→   427→                if escaped {
   428→   428→                        escaped = false
   429→   429→                        continue
   430→   430→                }
   431→   431→                if ch == '\\' {
   432→   432→                        escaped = true
   433→   433→                        continue
   434→   434→                }
   435→   435→                if ch == '"' {
   436→   436→                        inString = !inString
   437→   437→                        continue
   438→   438→                }
   439→   439→                if inString {
   440→   440→                        continue
   441→   441→                }
   442→   442→                switch ch {
   443→   443→                case '{':
   444→   444→                        depth++
   445→   445→                case '}':
   446→   446→                        depth--
   447→   447→                        if depth == 0 {
   448→   448→                                return response[start : i+1]
   449→   449→                        }
   450→   450→                }
   451→   451→        }
   452→   452→        return ""
   453→   453→}
   454→   454→
   455→   455→func formatConversation(conv []map[string]string) string {
   456→   456→        var b strings.Builder
   457→   457→        for _, msg := range conv {
   458→   458→                role := msg["role"]
   459→   459→                content := msg["content"]
   460→   460→                if len(content) > 4000 {
   461→   461→                        content = content[:4000] + "...(truncated)"
   462→   462→                }
   463→   463→                fmt.Fprintf(&b, "[%s]: %s\n\n", role, content)
   464→   464→        }
   465→   465→        return b.String()
   466→   466→}
   467→   467→
   468→   468→func compactConversation(conv []map[string]string, systemPrompt string) []map[string]string {
   469→   469→        if len(conv) <= 4 {
   470→   470→                return conv
   471→   471→        }
   472→   472→
   473→   473→        var findings []string
   474→   474→        for i := 2; i < len(conv)-2; i++ {
   475→   475→                msg := conv[i]

   561→   561→                        startLine = int(sl)

   564→   564→                        startLine = 1
   565→   565→                }

   567→   567→                if endLine > len(lines) {
   568→   568→                        endLine = len(lines)
   569→   569→                }

   571→   571→                        return ToolResult{Tool: "read_file", Success: false, Output: fmt.Sprintf("start_line %d exceeds file length %d", startLine, len(lines))}
   572→   572→                }
   573→   573→                var b strings.Builder
   574→   574→                fmt.Fprintf(&b, "%s (lines %d-%d of %d):\n", path, startLine, endLine, len(lines))

   582→   582→        })

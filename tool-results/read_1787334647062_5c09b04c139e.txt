   191→// --- mock Sentinel API ---
   192→
   193→// mockSentinelConfig configures a mockSentinel's behavior.
   194→type mockSentinelConfig struct {
   195→        // incidentStatus is returned by GET /incidents/{id}. If empty, the
   196→        // mock returns 404.
   197→        incidentStatus string
   198→        // incidentPRNumber, if non-nil, makes the mock return a PR. nil = no PR.
   199→        incidentPRNumber *int
   200→        // incidentSandboxVerified controls the sandbox_verified field.
   201→        incidentSandboxVerified bool
   202→        // incidentAutoMergeStatus controls the auto_merge_status field.
   203→        incidentAutoMergeStatus string
   204→        // triggerStatus is the HTTP status returned by POST /admin/investigate.
   205→        // 0 (default) = 200.
   206→        triggerStatus int
   207→        // pollAfter counts how many GET /incidents/{id} calls have been made.
   208→        // When non-zero, the mock returns incidentStatus only after this many
   209→        // polls; before that it returns "investigating". Lets us test the
   210→        // polling loop.
   211→        pollAfter int32
   212→}

   214→// mockSentinel returns a httptest.Server that emulates Sentinel's API
   215→// according to cfg. The returned server is already started; tests should
   216→// defer Close it.
   217→func mockSentinel(t *testing.T, cfg mockSentinelConfig) *httptest.Server {
   218→        t.Helper()
   219→        var pollCount int32
   220→        srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
   221→                switch {
   222→                case r.URL.Path == "/healthz":
   223→                        w.WriteHeader(200)
   224→                        _ = json.NewEncoder(w).Encode(map[string]any{
   225→                                "status":  "ok",
   226→                                "version": "test",
   227→                        })
   228→                case r.URL.Path == "/admin/investigate" && r.Method == http.MethodPost:
   229→                        if cfg.triggerStatus != 0 {
   230→                                w.WriteHeader(cfg.triggerStatus)
   231→                                _, _ = w.Write([]byte(`{"error":"mock trigger failure"}`))
   232→                                return
   233→                        }
   234→                        body, _ := io.ReadAll(r.Body)
   235→                        var req InvestigateRequest
   236→                        _ = json.Unmarshal(body, &req)
   237→                        w.WriteHeader(200)
   238→                        _ = json.NewEncoder(w).Encode(InvestigateResponse{
   239→                                Status:    "triggered",
   240→                                Service:   req.Service,
   241→                                ClusterID: "abc123def456",
   242→                        })
   243→                case strings.HasPrefix(r.URL.Path, "/incidents/"):
   244→                        id := strings.TrimPrefix(r.URL.Path, "/incidents/")
   245→                        if cfg.incidentStatus == "" {
   246→                                w.WriteHeader(404)
   247→                                _, _ = w.Write([]byte(`{"error":"not found"}`))
   248→                                return
   249→                        }
   250→                        // atomic.AddInt32 is safe to call concurrently — no mutex needed.
   251→                        polls := atomic.AddInt32(&pollCount, 1)
   252→                        status := cfg.incidentStatus
   253→                        if cfg.pollAfter > 0 && polls <= cfg.pollAfter {
   254→                                status = "investigating"
   255→                        }
   256→                        inc := Incident{
   257→                                IncidentID:      id,
   258→                                Status:          status,
   259→                                PRRepo:          "sentinel-bench/fixture",
   260→                                PRNumber:        cfg.incidentPRNumber,
   261→                                PRURL:           "https://github.com/sentinel-bench/fixture/pull/1",
   262→                                SandboxVerified: cfg.incidentSandboxVerified,
   263→                                AutoMergeStatus: cfg.incidentAutoMergeStatus,
   264→                        }
   265→                        w.WriteHeader(200)
   266→                        _ = json.NewEncoder(w).Encode(inc)
   267→                default:
   268→                        w.WriteHeader(404)
   269→                        _, _ = w.Write([]byte(`{"error":"unknown path"}`))
   270→                }
   271→        }))
   272→        return srv
   273→}

   472→// TestRunner_PollTimeout verifies that the runner stops polling after
   473→// the per-scenario budget elapses and records the last-seen incident.
   474→func TestRunner_PollTimeout(t *testing.T) {
   475→        dir := t.TempDir()
   476→        scenariosDir := scenarioFixture(t, dir, "T005", "fixture-poll-timeout")
   477→        resultsDir := filepath.Join(dir, "results")
   478→
   479→        srv := mockSentinel(t, mockSentinelConfig{
   480→                // Always "investigating" — never reaches terminal.
   481→                incidentStatus: "investigating",
   482→                // pollAfter=0 means: return the configured status immediately. So
   483→                // to force non-terminal, we set status to "investigating" itself
   484→                // — IsTerminal() returns false for "investigating".
   485→        })
   486→        defer srv.Close()
   487→
   488→        runner := newTestRunner(t, srv, scenariosDir, resultsDir, "")
   489→        runner.PollInterval = 5 * time.Millisecond
   490→        ctx := context.Background()
   491→        start := time.Now()
   492→        report, err := runner.Run(ctx, RunOptions{Timeout: 200 * time.Millisecond})
   493→        elapsed := time.Since(start)
   494→        if err != nil {
   495→                t.Fatalf("Run: %v", err)
   496→        }
   497→
   498→        s := report.Scenarios[0]
   499→        // Status is "fail" (no assertions passed) but NOT "error" — the
   500→        // poll timed out cleanly and we have a last-seen incident.
   501→        if s.Status == "error" {
   502→                t.Errorf("status: want fail or pass, got error (%s)", s.Error)
   503→        }
   504→        // The runner should have respected the timeout (within a generous
   505→        // tolerance — the polling loop checks the deadline between ticks).
   506→        if elapsed > 2*time.Second {
   507→                t.Errorf("runner took too long: %v (timeout was 200ms)", elapsed)
   508→        }
   509→}

   587→// TestEvaluateAssertions unit-tests the assertion evaluator directly.
   588→func TestEvaluateAssertions(t *testing.T) {
   589→        assertions := []Assertion{
   590→                {Type: "pr_opened", Weight: 0.25},
   591→                {Type: "pr_ci_green", Weight: 0.25},
   592→                {Type: "diff_similarity", Weight: 0.25, Threshold: 0.80},
   593→                {Type: "sentinel_automerge_approved", Weight: 0.25},
   594→        }
   595→        prNum := 1
   596→        inc := Incident{
   597→                Status:          "verified",
   598→                PRNumber:        &prNum,
   599→                SandboxVerified: true,
   600→                AutoMergeStatus: "approved",
   601→        }
   602→        prDiff := `--- a/foo
   603→+++ b/foo
   604→@@ -1,1 +1,1 @@
   605→-a
   606→+b
   607→`
   608→        gtDiff := prDiff // identical -> similarity 1.0
   609→
   610→        results := evaluateAssertions(assertions, inc, prDiff, gtDiff)
   611→        if len(results) != 4 {
   612→                t.Fatalf("want 4 results, got %d", len(results))
   613→        }
   614→        for _, r := range results {
   615→                if !r.Passed {
   616→                        t.Errorf("assertion %s: want pass, got fail (%s)", r.Type, r.Detail)
   617→                }
   618→        }
   619→
   620→        // diff_similarity assertion records observed + threshold.
   621→        var ds *AssertionResult
   622→        for i := range results {
   623→                if results[i].Type == "diff_similarity" {
   624→                        ds = &results[i]
   625→                        break
   626→                }
   627→        }
   628→        if ds == nil {
   629→                t.Fatal("no diff_similarity result")
   630→        }
   631→        if ds.Observed != 1.0 {
   632→                t.Errorf("observed: want 1.0, got %f", ds.Observed)
   633→        }
   634→        if ds.Threshold != 0.80 {
   635→                t.Errorf("threshold: want 0.80, got %f", ds.Threshold)
   636→        }
   637→}

   639→// TestScenarioScore verifies the weighted-sum scoring.
   640→func TestScenarioScore(t *testing.T) {
   641→        // 2 of 4 assertions pass with equal weight -> 0.5.
   642→        results := []AssertionResult{
   643→                {Type: "pr_opened", Weight: 0.25, Passed: true},
   644→                {Type: "pr_ci_green", Weight: 0.25, Passed: false},
   645→                {Type: "diff_similarity", Weight: 0.25, Passed: true},
   646→                {Type: "sentinel_automerge_approved", Weight: 0.25, Passed: false},
   647→        }
   648→        if got := scenarioScore(results); got != 0.5 {
   649→                t.Fatalf("score: want 0.5, got %f", got)
   650→        }
   651→
   652→        // All pass -> 1.0.
   653→        for i := range results {
   654→                results[i].Passed = true
   655→        }
   656→        if got := scenarioScore(results); got != 1.0 {
   657→                t.Fatalf("score (all pass): want 1.0, got %f", got)
   658→        }
   659→
   660→        // Empty -> 0 (avoids divide-by-zero).
   661→        if got := scenarioScore(nil); got != 0 {
   662→                t.Fatalf("score (empty): want 0, got %f", got)
   663→        }
   664→}

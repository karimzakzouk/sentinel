    58→// wiring holds concrete pointers to every initialized Sentinel component.
    59→// All adapter logic is expressed as methods on *wiring, so the wiring is
    60→// inspectable in a single place and there is no adapter-of-adapter
    61→// indirection.
    62→//
    63→// The mu field protects cfg and any other field that can be mutated by the
    64→// runtimecfg apply callback while investigation goroutines read it. Reads
    65→// from goroutines use RLock via Config(); the apply callback uses Lock via
    66→// mutateConfig().

    68→        mu  sync.RWMutex
    69→        cfg *config.Config
    70→

    77→        // the adapter from vectorStore on every config change.
    78→        agentsVS *agentsVectorStore
    79→

   139→// Config returns the current config under a read lock. Callers must not
   140→// mutate the returned struct; use mutateConfig for that.
   141→func (w *wiring) Config() *config.Config {
   142→        w.mu.RLock()
   143→        defer w.mu.RUnlock()
   144→        return w.cfg
   145→}
   146→
   147→// mutateConfig holds the write lock while fn mutates the config in place.
   148→// Used by the runtimecfg apply callback.
   149→func (w *wiring) mutateConfig(fn func(cfg *config.Config)) {
   150→        w.mu.Lock()
   151→        defer w.mu.Unlock()
   152→        fn(w.cfg)
   153→}

   157→// FindIncidentByPR satisfies chatops.Store. Loads the incident, then maps
   158→// the schemas types into chatops's flat Incident view.
   159→func (w *wiring) FindIncidentByPR(ctx context.Context, repo string, prNumber int) (*chatops.Incident, error) {
   160→        inc, err := w.store.FindIncidentByPR(ctx, repo, prNumber)
   161→        if err != nil || inc == nil {
   162→                return nil, err
   163→        }
   164→        out := &chatops.Incident{
   165→                ID:        inc.IncidentID.String(),
   166→                Service:   inc.Cluster.Service,
   167→                Namespace: inc.Cluster.Namespace,
   168→                Status:    string(inc.Status),
   169→        }
   170→        if inc.State.RootCause != nil {
   171→                out.RootCause = inc.State.RootCause.Statement
   172→                out.Confidence = inc.State.RootCause.Confidence
   173→                out.HasSchemaMigration = inc.State.RootCause.HasSchemaMigration
   174→        }
   175→        if inc.State.FixProposal != nil {
   176→                out.FixType = string(inc.State.FixProposal.FixType)
   177→                if inc.State.FixProposal.BaseBranch != nil {
   178→                        out.BaseBranch = *inc.State.FixProposal.BaseBranch
   179→                }
   180→                if inc.State.FixProposal.BranchName != nil {
   181→                        out.PRBranch = *inc.State.FixProposal.BranchName
   182→                }
   183→        }
   184→        if inc.State.ExecutionResult != nil {
   185→                if inc.State.ExecutionResult.PRURL != nil {
   186→                        out.PRURL = *inc.State.ExecutionResult.PRURL
   187→                }
   188→                if inc.State.ExecutionResult.PRNumber != nil {
   189→                        out.PRNumber = *inc.State.ExecutionResult.PRNumber
   190→                }
   191→                out.SandboxVerified = inc.State.ExecutionResult.SandboxVerified
   192→                if inc.State.ExecutionResult.SandboxSummary != nil {
   193→                        out.SandboxSummary = *inc.State.ExecutionResult.SandboxSummary
   194→                }
   195→        }
   196→        out.PRRepo = repo
   197→        out.AgentTrace = inc.State.AgentTrace
   198→        for _, h := range inc.State.Hypotheses {
   199→                out.Hypotheses = append(out.Hypotheses, chatops.Hypothesis{
   200→                        ID:         h.HID,
   201→                        Statement:  h.Statement,
   202→                        Confidence: h.Confidence,
   203→                        Falsified:  h.Falsified,
   204→                })
   205→        }
   206→        if inc.State.GovernanceDecision != nil {
   207→                out.PolicyVerdict = string(inc.State.GovernanceDecision.Verdict)
   208→        }
   209→        return out, nil
   210→}

   419→// SimilarIncidents satisfies admission.IncidentMemory. Embeds the manifest
   420→// JSON and asks the vector store for nearest neighbours above threshold.
   421→func (w *wiring) SimilarIncidents(ctx context.Context, obj admission.Manifest, threshold float64) ([]string, error) {
   422→        if w.vectorStore == nil {
   423→                return nil, nil
   424→        }
   425→        metas, scores, err := w.vectorStore.Search(ctx, string(obj.JSON), 5)
   426→        if err != nil {
   427→                return nil, err
   428→        }
   429→        var out []string
   430→        for i, m := range metas {
   431→                if i >= len(scores) {
   432→                        break
   433→                }
   434→                if float64(scores[i]) >= threshold {
   435→                        out = append(out, m.IncidentID)
   436→                }
   437→        }
   438→        return out, nil
   439→}

   565→// HandleAlert satisfies alertmanager.SignalHandler. Looks up the GitHub
   566→// repo via K8s annotations when the alert payload didn't carry one, then
   567→// forwards the alert to the detection clusterer as a SignalLogError signal.
   568→func (w *wiring) HandleAlert(service, namespace, reason, repo, description string, _ map[string]string) {
   569→        ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
   570→        defer cancel()
   571→
   572→        if repo == "" && w.k8sClient != nil {
   573→                annotations, _ := w.k8sClient.LookupWorkloadAnnotations(ctx, service, namespace)
   574→                if annotations != nil {
   575→                        repo = annotations["sentinel.io/repo"]
   576→                }
   577→        }
   578→
   579→        chatopsLogInfo("alertmanager → clusterer",
   580→                "service", service,
   581→                "namespace", namespace,
   582→                "reason", reason,
   583→                "repo", repo,
   584→        )
   585→
   586→        w.clusterer.Send(ctx, detection.Signal{
   587→                Type:      detection.SignalLogError,
   588→                Service:   service,
   589→                Namespace: namespace,
   590→                Reason:    reason,
   591→                Repo:      repo,
   592→                Timestamp: time.Now(),
   593→        })
   594→}

   603→// agentsVectorStore adapts *feedback.VectorStore to agents.VectorStore.
   604→// Defined as a separate type because agents.VectorStore and

   606→// return types — Go cannot express that on a single receiver.
   607→type agentsVectorStore struct {
   608→        vs *feedback.VectorStore

   611→// Search satisfies agents.VectorStore.
   612→func (a *agentsVectorStore) Search(ctx context.Context, text string, k int) ([]*agents.VectorMetadata, []float32, error) {
   613→        metas, scores, err := a.vs.Search(ctx, text, k)
   614→        if err != nil || len(metas) == 0 {
   615→                return nil, nil, err
   616→        }
   617→        out := make([]*agents.VectorMetadata, len(metas))
   618→        for i, m := range metas {
   619→                out[i] = &agents.VectorMetadata{
   620→                        IncidentID: m.IncidentID,
   621→                        Service:    m.Service,
   622→                        RootCause:  m.RootCause,
   623→                        FixType:    m.FixType,
   624→                        FixPattern: m.FixPattern,
   625→                        Outcome:    m.Outcome,
   626→                }
   627→        }
   628→        return out, scores, nil
   629→}

    95→    95→// handleIncident returns a single incident by full UUID or short prefix.
    96→    96→func (s *Server) handleIncident(w http.ResponseWriter, r *http.Request) {
    97→    97→        idStr := strings.TrimPrefix(r.URL.Path, "/incidents/")
    98→    98→
    99→    99→        id, err := uuid.Parse(idStr)
   100→   100→        if err == nil {
   101→   101→                // Full UUID — direct lookup.
   102→   102→                incident, err := s.store.GetIncident(r.Context(), id)
   103→   103→                if err != nil {
   104→   104→                        writeJSON(w, 500, map[string]string{"error": err.Error()})
   105→   105→                        return
   106→   106→                }
   107→   107→                if incident == nil {
   108→   108→                        writeJSON(w, 404, map[string]string{"error": "incident not found"})
   109→   109→                        return
   110→   110→                }
   111→   111→                enrichIncidentPRFields(incident)
   112→   112→                writeJSON(w, 200, incident)
   113→   113→                return
   114→   114→        }
   115→   115→
   116→   116→        // Not a full UUID — try prefix matching. FindIncidentByPrefix requires
   117→   117→        // ≥4 chars to prevent accidental collisions on short prefixes.
   118→   118→        incident, err := s.store.FindIncidentByPrefix(r.Context(), idStr)
   119→   119→        if err != nil {
   120→   120→                writeJSON(w, 500, map[string]string{"error": err.Error()})
   121→   121→                return
   122→   122→        }
   123→   123→        if incident == nil {
   124→   124→                writeJSON(w, 404, map[string]string{"error": "no incident matches that ID prefix (need at least 4 chars)"})
   125→   125→                return
   126→   126→        }
   127→   127→        enrichIncidentPRFields(incident)
   128→   128→        writeJSON(w, 200, incident)
   129→   129→}

   131→   131→// enrichIncidentPRFields mirrors pr_number/pr_url/pr_branch +
   132→   132→// sandbox_summary + sandbox_verified from the nested state.execution_result
   133→   133→// (which is what the agents write to) onto the top-level Incident fields
   134→   134→// (which is what the frontend reads). Keeps the JSON contract stable
   135→   135→// without rewriting every SELECT.
   136→   136→func enrichIncidentPRFields(inc *schemas.Incident) {
   137→   137→        if inc == nil {
   138→   138→                return
   139→   139→        }
   140→   140→        if er := inc.State.ExecutionResult; er != nil {
   141→   141→                if inc.PRNumber == nil && er.PRNumber != nil {
   142→   142→                        inc.PRNumber = er.PRNumber
   143→   143→                }
   144→   144→                if inc.PRURL == "" && er.PRURL != nil {
   145→   145→                        inc.PRURL = *er.PRURL
   146→   146→                }
   147→   147→                if inc.PRBranch == "" && er.PRBranch != nil {
   148→   148→                        inc.PRBranch = *er.PRBranch
   149→   149→                }
   150→   150→                if inc.SandboxSummary == "" && er.SandboxSummary != nil {
   151→   151→                        inc.SandboxSummary = *er.SandboxSummary
   152→   152→                }
   153→   153→                if !inc.SandboxVerified && er.SandboxVerified {
   154→   154→                        inc.SandboxVerified = er.SandboxVerified
   155→   155→                }
   156→   156→        }
   157→   157→        if fp := inc.State.FixProposal; fp != nil {
   158→   158→                if inc.FixType == "" {
   159→   159→                        inc.FixType = string(fp.FixType)
   160→   160→                }
   161→   161→        }
   162→   162→}

   202→   202→// handleInvestigate is the manual-trigger endpoint. POST with a JSON body

   205→   205→        if r.Method != http.MethodPost {
   206→   206→                writeJSON(w, 405, map[string]string{"error": "method not allowed"})

   214→   214→                PodLogs   string `json:"pod_logs"`
   215→   215→        }

   217→   217→        if err := json.NewDecoder(r.Body).Decode(&spec); err != nil {

   219→   219→                return

   221→   221→        if spec.Service == "" {
   222→   222→                writeJSON(w, 400, map[string]string{"error": "service is required"})
   223→   223→                return
   224→   224→        }
   225→   225→        if spec.Namespace == "" {
   226→   226→                spec.Namespace = "default"
   227→   227→        }
   228→   228→        if spec.Reason == "" {
   229→   229→                spec.Reason = "manual trigger"
   230→   230→        }

   232→   232→                clusterID, err := s.investigateWithLogs(spec.Service, spec.Namespace, spec.Reason, spec.Repo, spec.PodLogs)
   233→   233→                if err != nil {
   234→   234→                        writeJSON(w, 500, map[string]string{"error": err.Error()})
   235→   235→                        return
   236→   236→                }

   238→   238→                return

   240→   240→        clusterID, err := s.investigate(spec.Service, spec.Namespace, spec.Reason)
   241→   241→        if err != nil {
   242→   242→                writeJSON(w, 500, map[string]string{"error": err.Error()})
   243→   243→                return
   244→   244→        }

   246→   246→                "status":     "triggered",
   247→   247→                "service":    spec.Service,
   248→   248→                "cluster_id": clusterID,

   298→   298→// maxSSESubscribers caps concurrent SSE connections to prevent memory
   299→   299→// exhaustion DoS. SSE is unauthenticated (when apiToken is empty) and
   300→   300→// each subscriber pins a goroutine + 16-entry buffered channel + map
   301→   301→// entry (~10KB). 100 subscribers is generous for a single-pod dashboard.
   302→   302→const maxSSESubscribers = 100
   303→   303→
   304→   304→// handleSSE streams events to the client. Subscribes to the broadcaster,
   305→   305→// writes "data: {...}\n\n" frames, and flushes on each event. A 15s
   306→   306→// keep-alive ping prevents intermediate proxies from closing the connection.
   307→   307→func (s *Server) handleSSE(w http.ResponseWriter, r *http.Request) {
   308→   308→        w.Header().Set("Content-Type", "text/event-stream")
   309→   309→        w.Header().Set("Cache-Control", "no-cache")
   310→   310→        w.Header().Set("Connection", "keep-alive")
   311→   311→        w.Header().Set("Access-Control-Allow-Origin", "*")
   312→   312→
   313→   313→        flusher, ok := w.(http.Flusher)
   314→   314→        if !ok {
   315→   315→                http.Error(w, "streaming not supported", http.StatusInternalServerError)
   316→   316→                return
   317→   317→        }
   318→   318→
   319→   319→        s.eventSubsMu.Lock()
   320→   320→        if len(s.eventSubs) >= maxSSESubscribers {
   321→   321→                s.eventSubsMu.Unlock()
   322→   322→                http.Error(w, "too many SSE subscribers (max "+strconv.Itoa(maxSSESubscribers)+")", http.StatusServiceUnavailable)
   323→   323→                return
   324→   324→        }
   325→   325→        ch := make(chan ServerEvent, 16)
   326→   326→        s.eventSubs[ch] = true
   327→   327→        s.eventSubsMu.Unlock()
   328→   328→        defer func() {
   329→   329→                s.eventSubsMu.Lock()
   330→   330→                delete(s.eventSubs, ch)
   331→   331→                s.eventSubsMu.Unlock()
   332→   332→        }()
   333→   333→
   334→   334→        // Send events as unnamed SSE messages (no `event:` line) so the
   335→   335→        // frontend's EventSource.onmessage handler fires for every event.
   336→   336→        // The event type is embedded in the JSON payload's `type` field.
   337→   337→        fmt.Fprintf(w, "data: {\"type\":\"connected\",\"status\":\"connected\"}\n\n")
   338→   338→        flusher.Flush()
   339→   339→
   340→   340→        ticker := time.NewTicker(15 * time.Second)
   341→   341→        defer ticker.Stop()
   342→   342→
   343→   343→        for {
   344→   344→                select {
   345→   345→                case <-r.Context().Done():
   346→   346→                        return
   347→   347→                case event := <-ch:
   348→   348→                        data, _ := json.Marshal(event)
   349→   349→                        fmt.Fprintf(w, "data: %s\n\n", data)
   350→   350→                        flusher.Flush()
   351→   351→                case <-ticker.C:
   352→   352→                        fmt.Fprintf(w, ": ping\n\n")
   353→   353→                        flusher.Flush()
   354→   354→                }
   355→   355→        }
   356→   356→}

   358→   358→// handleWebhook dispatches /webhook/github to the prreview + chatops
   359→   359→// handlers. The body is buffered once so both handlers can read it; each
   360→   360→// handler ignores event types it doesn't care about.
   361→   361→func (s *Server) handleWebhook(w http.ResponseWriter, r *http.Request) {
   362→   362→        if s.webhookHandler == nil && s.chatopsHandler == nil {
   363→   363→                writeJSON(w, 503, map[string]string{"error": "webhook handler not configured"})
   364→   364→                return
   365→   365→        }
   366→   366→
   367→   367→        bodyBytes, err := io.ReadAll(io.LimitReader(r.Body, 1<<20)) // 1MB cap.
   368→   368→        if err != nil {
   369→   369→                writeJSON(w, 400, map[string]string{"error": "read body failed"})
   370→   370→                return
   371→   371→        }
   372→   372→        multiReader := func() io.Reader { return bytes.NewReader(bodyBytes) }
   373→   373→
   374→   374→        // 1. Dispatch to prreview (predictive PR review — pull_request events).
   375→   375→        if s.webhookHandler != nil {
   376→   376→                req := r.Clone(r.Context())
   377→   377→                req.Body = io.NopCloser(multiReader())
   378→   378→                s.webhookHandler.ServeHTTP(w, req)
   379→   379→        }
   380→   380→
   381→   381→        // 2. Dispatch to chatops (issue_comment events for /sentinel commands).
   382→   382→        // Short-circuit for non-issue_comment events so we don't buffer +
   383→   383→        // parse for events chatops doesn't care about.
   384→   384→        if s.chatopsHandler != nil {
   385→   385→                eventType := r.Header.Get("X-GitHub-Event")
   386→   386→                if eventType == "issue_comment" {
   387→   387→                        req := r.Clone(r.Context())
   388→   388→                        req.Body = io.NopCloser(multiReader())
   389→   389→                        s.chatopsHandler.ServeHTTP(w, req)
   390→   390→                }
   391→   391→        }
   392→   392→}

   413→   413→// handleGitHubApp manages the GitHub App config.
   414→   414→// GET    → status (configured? app_id? slug? install_id?).
   415→   415→// POST   → store App ID + PEM + optional install ID + webhook secret.
   416→   416→// DELETE → remove App config (fall back to PAT).
   417→   417→func (s *Server) handleGitHubApp(w http.ResponseWriter, r *http.Request) {
   418→   418→        if s.githubAppManager == nil {
   419→   419→                writeJSON(w, 503, map[string]string{"error": "github app manager not configured"})
   420→   420→                return
   421→   421→        }
   422→   422→
   423→   423→        switch r.Method {
   424→   424→        case http.MethodGet:
   425→   425→                status := s.githubAppManager.GetGitHubAppStatus()
   426→   426→                writeJSON(w, 200, status)
   427→   427→
   428→   428→        case http.MethodPost:
   429→   429→                r.Body = http.MaxBytesReader(nil, r.Body, maxRequestBodyBytes)
   430→   430→                var req struct {
   431→   431→                        AppID         string `json:"app_id"`
   432→   432→                        PrivateKey    string `json:"private_key"`
   433→   433→                        InstallID     string `json:"install_id"`
   434→   434→                        WebhookSecret string `json:"webhook_secret"`
   435→   435→                }
   436→   436→                if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   437→   437→                        writeJSON(w, 400, map[string]string{"error": "invalid JSON: " + err.Error()})
   438→   438→                        return
   439→   439→                }
   440→   440→                if req.AppID == "" || req.PrivateKey == "" {
   441→   441→                        writeJSON(w, 400, map[string]string{"error": "app_id and private_key are required"})
   442→   442→                        return
   443→   443→                }
   444→   444→                slug, err := s.githubAppManager.SetGitHubApp(req.AppID, req.PrivateKey, req.InstallID, req.WebhookSecret)
   445→   445→                if err != nil {
   446→   446→                        writeJSON(w, 500, map[string]string{"error": err.Error()})
   447→   447→                        return
   448→   448→                }
   449→   449→                logging.Info("api: GitHub App configured", "app_id", req.AppID, "slug", slug)
   450→   450→                writeJSON(w, 200, map[string]string{
   451→   451→                        "status": "configured",
   452→   452→                        "slug":   slug,
   453→   453→                        "bot":    slug + "[bot]",
   454→   454→                        "hint":   "If install_id was empty, run `sentinelctl app set-install <id>` after installing the App.",
   455→   455→                })
   456→   456→
   457→   457→        case http.MethodDelete:
   458→   458→                if err := s.githubAppManager.RemoveGitHubApp(); err != nil {

   466→   466→                })

   473→   473→// handleGitHubAppInstall sets just the installation ID. Called separately
   474→   474→// from the App credentials because the install ID isn't known until the
   475→   475→// user installs the App on their repo.
   476→   476→func (s *Server) handleGitHubAppInstall(w http.ResponseWriter, r *http.Request) {
   477→   477→        if s.githubAppManager == nil {
   478→   478→                writeJSON(w, 503, map[string]string{"error": "github app manager not configured"})
   479→   479→                return
   480→   480→        }
   481→   481→        if r.Method != http.MethodPost {
   482→   482→                writeJSON(w, 405, map[string]string{"error": "method not allowed"})
   483→   483→                return
   484→   484→        }
   485→   485→        var req struct {
   486→   486→                InstallID string `json:"install_id"`
   487→   487→        }
   488→   488→        if err := json.NewDecoder(http.MaxBytesReader(nil, r.Body, maxRequestBodyBytes)).Decode(&req); err != nil {
   489→   489→                writeJSON(w, 400, map[string]string{"error": "invalid JSON: " + err.Error()})
   490→   490→                return
   491→   491→        }
   492→   492→        if req.InstallID == "" {
   493→   493→                writeJSON(w, 400, map[string]string{"error": "install_id is required"})
   494→   494→                return
   495→   495→        }
   496→   496→        if err := s.githubAppManager.SetGitHubAppInstall(req.InstallID); err != nil {
   497→   497→                writeJSON(w, 500, map[string]string{"error": err.Error()})
   498→   498→                return
   499→   499→        }
   500→   500→        logging.Info("api: GitHub App install ID set", "install_id", req.InstallID)
   501→   501→        writeJSON(w, 200, map[string]string{"status": "configured", "install_id": req.InstallID})
   502→   502→}

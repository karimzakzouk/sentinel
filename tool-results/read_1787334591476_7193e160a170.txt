   370→// ResolvedBaseURL returns the LLM base URL, deriving from the provider name
   371→// when LLMBaseURL is unset.
   372→func (c *Config) ResolvedBaseURL() string {
   373→        if c.LLMBaseURL != "" {
   374→                return c.LLMBaseURL
   375→        }
   376→        switch strings.ToLower(c.LLMProvider) {
   377→        case "openai":
   378→                return "https://api.openai.com/v1"
   379→        case "zai":
   380→                return "https://api.z.ai/api/paas/v4"
   381→        case "gemini":
   382→                return "https://generativelanguage.googleapis.com/v1beta/openai"
   383→        case "ollama":
   384→                return "http://ollama:11434/v1"
   385→        case "vllm":
   386→                return "http://vllm:8000/v1"
   387→        case "litellm":
   388→                return "http://litellm:4000/v1"
   389→        default:
   390→                return "https://api.z.ai/api/paas/v4"
   391→        }
   392→}

   394→// CanWriteTo reports whether Sentinel is allowed to apply fixes in this namespace.

   396→        if len(c.WriteNamespaces) == 0 {

   407→// IsRepoAllowed reports whether repo is in the AllowedGitHubRepos allowlist.
   408→// An empty allowlist allows all repos (NOT recommended for production).
   409→// Matching is case-insensitive and ignores a trailing ".git".
   410→func (c *Config) IsRepoAllowed(repo string) bool {
   411→        if len(c.AllowedGitHubRepos) == 0 {
   412→                return true
   413→        }
   414→        target := normalizeRepoForCompare(repo)
   415→        for _, allowed := range c.AllowedGitHubRepos {
   416→                if normalizeRepoForCompare(allowed) == target {
   417→                        return true
   418→                }
   419→        }
   420→        return false
   421→}

   423→// Validate returns a list of blocking errors and a list of non-blocking
   424→// warnings. A non-empty errors slice means Sentinel cannot start.
   425→func (c *Config) Validate() (errors []string, warnings []string) {
   426→        if c.LLMAPIKey == "" {
   427→                errors = append(errors, "SENTINEL_LLM_API_KEY is empty — Sentinel cannot run without an LLM API key")
   428→        }
   429→        if c.LLMProvider == "" {
   430→                errors = append(errors, "SENTINEL_LLM_PROVIDER is empty")
   431→        } else if !isKnownProvider(c.LLMProvider) {
   432→                warnings = append(warnings, fmt.Sprintf(
   433→                        "SENTINEL_LLM_PROVIDER=%q is not in the known-providers list — "+
   434→                                "if this is a typo, LLM calls will be sent to the wrong endpoint with your API key attached",
   435→                        c.LLMProvider))
   436→        }
   437→        if c.LLMFastModel == "" {
   438→                errors = append(errors, "SENTINEL_LLM_FAST_MODEL is empty")
   439→        }
   440→        if c.LLMFrontierModel == "" {
   441→                errors = append(errors, "SENTINEL_LLM_FRONTIER_MODEL is empty")
   442→        }
   443→
   444→        if c.GitHubToken == "" {
   445→                errors = append(errors, "SENTINEL_GITHUB_TOKEN is empty — Sentinel needs a GitHub PAT to clone repos and open PRs")
   446→        }
   447→
   448→        if len(c.WriteNamespaces) == 0 {
   449→                warnings = append(warnings, "SENTINEL_WRITE_NAMESPACES is empty — Sentinel cannot write fixes to any namespace")
   450→        }
   451→
   452→        if c.APIToken == "" {
   453→                warnings = append(warnings, "SENTINEL_API_TOKEN is empty — admin API is unauthenticated. Set this in production.")
   454→        }
   455→
   456→        if len(c.AllowedGitHubRepos) == 0 && c.GitHubToken != "" {
   457→                warnings = append(warnings, "SENTINEL_ALLOWED_GITHUB_REPOS is empty — Sentinel will clone ANY repo annotated on a pod using the configured GitHub PAT. Set an explicit allowlist.")
   458→        }
   459→
   460→        if c.SlackBotToken != "" && c.SlackSigningSecret == "" {
   461→                warnings = append(warnings, "SENTINEL_SLACK_SIGNING_SECRET is empty but SLACK_BOT_TOKEN is set — Slack requests will be rejected")
   462→        }
   463→
   464→        if c.SandboxMaxRetries < 0 {
   465→                warnings = append(warnings, fmt.Sprintf("SENTINEL_SANDBOX_MAX_RETRIES=%d (negative) — clamping to 0", c.SandboxMaxRetries))
   466→                c.SandboxMaxRetries = 0
   467→        }
   468→        if c.SandboxMaxRetries > 10 {
   469→                warnings = append(warnings, fmt.Sprintf("SENTINEL_SANDBOX_MAX_RETRIES=%d (very high) — each retry takes ~5-10min", c.SandboxMaxRetries))
   470→        }
   471→        if c.InvestigatorTimeoutSec <= 0 {
   472→                warnings = append(warnings, fmt.Sprintf("SENTINEL_INVESTIGATOR_TIMEOUT=%d (non-positive)", c.InvestigatorTimeoutSec))
   473→        }
   474→        if c.ConfidenceAutonomous < 0 || c.ConfidenceAutonomous > 1 {
   475→                warnings = append(warnings, fmt.Sprintf("SENTINEL_CONFIDENCE_THRESHOLD_AUTONOMOUS=%.2f out of [0,1] — clamping", c.ConfidenceAutonomous))
   476→                c.ConfidenceAutonomous = clamp01(c.ConfidenceAutonomous)
   477→        }
   478→
   479→        return errors, warnings
   480→}

   484→                return 0
   485→        }
   486→        if f > 1 {
   487→                return 1
   488→        }
   489→        return f
   490→}

   545→func getenvBool(key string, def bool) bool {
   546→        v := os.Getenv(key)
   547→        if v == "" {
   548→                return def
   549→        }
   550→        switch strings.ToLower(v) {
   551→        case "1", "true", "yes", "on":

   554→                return false
   555→        }
   556→        return def

   563→func splitCSV(s string) []string {
   564→        if s == "" {
   565→                return nil
   566→        }
   567→        parts := strings.Split(s, ",")
   568→        out := make([]string, 0, len(parts))
   569→        for _, p := range parts {
   570→                p = strings.TrimSpace(p)
   571→                if p != "" {
   572→                        out = append(out, p)
   573→                }
   574→        }
   575→        return out
   576→}

   578→// loadLLMProviders parses SENTINEL_LLM_PROVIDERS as a JSON array of
   579→// ProviderConfig. Entries missing ProviderType or APIKey (for non-local
   580→// providers) are dropped. Returns nil on parse failure (logs a warning).
   581→func loadLLMProviders() []schemas.ProviderConfig {
   582→        raw := os.Getenv("SENTINEL_LLM_PROVIDERS")
   583→        if raw == "" {
   584→                return nil
   585→        }
   586→        var configs []schemas.ProviderConfig
   587→        if err := json.Unmarshal([]byte(raw), &configs); err != nil {
   588→                logging.Error("SENTINEL_LLM_PROVIDERS failed to parse as JSON — ignoring", "err", err)
   589→                return nil
   590→        }
   591→        valid := make([]schemas.ProviderConfig, 0, len(configs))
   592→        for _, c := range configs {
   593→                if c.ProviderType == "" {
   594→                        logging.Info("LLM provider entry missing 'provider' field — skipping", "id", c.ID)
   595→                        continue
   596→                }
   597→                if c.APIKey == "" && c.ProviderType != "ollama" && c.ProviderType != "vllm" {
   598→                        logging.Info("LLM provider has no apiKey — skipping", "id", c.ID, "provider", c.ProviderType)
   599→                        continue
   600→                }
   601→                valid = append(valid, c)
   602→        }
   603→        return valid
   604→}

    53→// fakePauseController records Pause/Resume calls for assertions.
    54→type fakePauseController struct {
    55→        mu          sync.Mutex
    56→        paused      bool
    57→        pauseDur    time.Duration
    58→        pauseReason string
    59→}
    60→
    61→func (f *fakePauseController) IsPaused() bool {
    62→        f.mu.Lock()
    63→        defer f.mu.Unlock()
    64→        return f.paused
    65→}

    67→func (f *fakePauseController) Pause(d time.Duration, reason string) {
    68→        f.mu.Lock()
    69→        defer f.mu.Unlock()
    70→        f.paused = true
    71→        f.pauseDur = d
    72→        f.pauseReason = reason
    73→}

    75→func (f *fakePauseController) Resume() {
    76→        f.mu.Lock()
    77→        defer f.mu.Unlock()
    78→        f.paused = false

   150→// TestServer_BearerAuth verifies the bearer-token middleware:
   151→//   - With apiToken set, admin endpoints require 'Authorization: Bearer <token>'.
   152→//   - Missing token → 401; wrong token → 401; correct token → 200.
   153→//   - Public endpoints (healthz, readyz, metrics, /) are always allowed.
   154→func TestServer_BearerAuth(t *testing.T) {
   155→        const token = "secret-token"
   156→
   157→        cases := []struct {
   158→                name       string
   159→                tokenSet   bool
   160→                path       string
   161→                method     string
   162→                authHeader string
   163→                wantCode   int
   164→        }{
   165→                {"no token configured — admin allowed", false, "/admin/pause", "POST", "", 200},
   166→                {"token set — no auth header", true, "/admin/pause", "POST", "", 401},
   167→                {"token set — wrong token", true, "/admin/pause", "POST", "Bearer wrong", 401},
   168→                {"token set — malformed header", true, "/admin/pause", "POST", "Token xyz", 401},
   169→                {"token set — correct token", true, "/admin/pause", "POST", "Bearer " + token, 200},
   170→                {"token set — healthz is public", true, "/healthz", "GET", "", 200},
   171→                {"token set — readyz is public", true, "/readyz", "GET", "", 200},
   172→                {"token set — metrics is public", true, "/metrics", "GET", "", 200},
   173→                {"token set — / is public", true, "/", "GET", "", 200},
   174→                {"token set — GET /incidents is read-only", true, "/incidents", "GET", "", 200},
   175→                {"token set — GET /providers is read-only", true, "/providers", "GET", "", 200},
   176→                {"token set — GET /settings is read-only", true, "/settings", "GET", "", 200},
   177→                {"token set — POST /settings requires token (none)", true, "/settings", "POST", "", 401},
   178→                {"token set — POST /settings with token", true, "/settings", "POST", "Bearer " + token, 200},
   179→        }
   180→
   181→        for _, tc := range cases {
   182→                t.Run(tc.name, func(t *testing.T) {
   183→                        s := newTestServer()
   184→                        if tc.tokenSet {
   185→                                s.SetAPIToken(token)
   186→                        }
   187→                        // POST /settings needs a runtime cfg saver to return 200.
   188→                        s.SetRuntimeCfgSaver(&noopCfgSaver{})
   189→                        // /admin/pause needs a pause controller to return 200.
   190→                        s.SetPauseController(&fakePauseController{})
   191→
   192→                        var body string
   193→                        if tc.method == "POST" {
   194→                                switch tc.path {
   195→                                case "/admin/pause":
   196→                                        body = `{}`
   197→                                case "/settings":
   198→                                        body = `{}`
   199→                                default:
   200→                                        body = `{}`
   201→                                }
   202→                        }
   203→                        req := httptest.NewRequest(tc.method, tc.path, strings.NewReader(body))
   204→                        if tc.authHeader != "" {
   205→                                req.Header.Set("Authorization", tc.authHeader)
   206→                        }
   207→                        w := httptest.NewRecorder()
   208→                        s.http.Handler.ServeHTTP(w, req)
   209→
   210→                        if w.Code != tc.wantCode {
   211→                                t.Errorf("got %d, want %d (body: %s)", w.Code, tc.wantCode, w.Body.String())
   212→                        }
   213→                })
   214→        }
   215→}

   222→// TestServer_SSEBroadcaster verifies that events flow from BroadcastEvent
   223→// to connected SSE subscribers. We test the broadcaster mechanism directly
   224→// rather than via an HTTP round-trip because httptest.ResponseRecorder
   225→// does not support concurrent reads while the handler is still writing.
   226→func TestServer_SSEBroadcaster(t *testing.T) {
   227→        s := newTestServer()
   228→
   229→        // Register a subscriber directly (same as handleSSE does).
   230→        ch := make(chan ServerEvent, 16)
   231→        s.eventSubsMu.Lock()
   232→        s.eventSubs[ch] = true
   233→        s.eventSubsMu.Unlock()
   234→        defer func() {
   235→                s.eventSubsMu.Lock()
   236→                delete(s.eventSubs, ch)
   237→                s.eventSubsMu.Unlock()
   238→        }()
   239→
   240→        // Broadcast an event.
   241→        s.BroadcastEvent("incident.created", map[string]string{"id": "abc"})
   242→
   243→        select {
   244→        case ev := <-ch:
   245→                if ev.Type != "incident.created" {
   246→                        t.Errorf("event Type = %q, want incident.created", ev.Type)
   247→                }
   248→                data, ok := ev.Data.(map[string]string)
   249→                if !ok {
   250→                        t.Fatalf("event Data type = %T, want map[string]string", ev.Data)
   251→                }
   252→                if data["id"] != "abc" {
   253→                        t.Errorf("event Data[id] = %q, want abc", data["id"])
   254→                }
   255→        case <-time.After(time.Second):
   256→                t.Fatal("did not receive broadcast event within 1s")
   257→        }
   258→}

   295→// TestServer_SSEBroadcasterSlowSubDropped verifies a slow subscriber (full
   296→// channel) is dropped via the non-blocking select rather than blocking
   297→// the broadcaster.
   298→func TestServer_SSEBroadcasterSlowSubDropped(t *testing.T) {
   299→        s := newTestServer()
   300→
   301→        // Register a subscriber with a tiny buffer that we never drain.
   302→        ch := make(chan ServerEvent, 1)
   303→        s.eventSubsMu.Lock()
   304→        s.eventSubs[ch] = true
   305→        s.eventSubsMu.Unlock()
   306→        defer func() {
   307→                s.eventSubsMu.Lock()
   308→                delete(s.eventSubs, ch)
   309→                s.eventSubsMu.Unlock()
   310→        }()
   311→
   312→        // Broadcast 5 events. Only the first fits in the buffer; the rest are
   313→        // dropped via the non-blocking select in BroadcastEvent.
   314→        for i := 0; i < 5; i++ {
   315→                s.BroadcastEvent("e", i)
   316→        }
   317→
   318→        // Drain: we should get at most 1 event (the first one). The broadcaster
   319→        // must not have blocked.
   320→        received := 0
   321→loop:
   322→        for {
   323→                select {
   324→                case <-ch:
   325→                        received++
   326→                default:
   327→                        break loop
   328→                }
   329→        }
   330→        if received > 1 {
   331→                t.Errorf("expected at most 1 event in the buffer, got %d", received)
   332→        }
   333→}

   434→// TestServer_APIMeta verifies /api/meta returns the endpoint list and the
   435→// chatops command list (so the frontend can render the chatops section
   436→// without hardcoding commands).
   437→func TestServer_APIMeta(t *testing.T) {
   438→        s := newTestServer()
   439→        req := httptest.NewRequest("GET", "/api/meta", nil)
   440→        w := httptest.NewRecorder()
   441→        s.http.Handler.ServeHTTP(w, req)
   442→        if w.Code != 200 {
   443→                t.Fatalf("got %d, want 200", w.Code)
   444→        }
   445→        var resp map[string]any
   446→        if err := json.Unmarshal(w.Body.Bytes(), &resp); err != nil {
   447→                t.Fatalf("unmarshal: %v", err)
   448→        }
   449→        endpoints, _ := resp["endpoints"].([]any)
   450→        if len(endpoints) == 0 {
   451→                t.Error("expected non-empty endpoints list")
   452→        }
   453→        // /api/meta should be in the list (self-reference).
   454→        found := false
   455→        for _, e := range endpoints {
   456→                if e == "/api/meta" {
   457→                        found = true
   458→                        break
   459→                }
   460→        }
   461→        if !found {
   462→                t.Error("/api/meta not in endpoints list")
   463→        }
   464→        // The chatops command list must be present and non-empty — the registry
   465→        // is populated at init() time from internal/chatops/commands*.go.
   466→        commands, _ := resp["commands"].([]any)
   467→        if len(commands) == 0 {
   468→                t.Fatal("expected non-empty commands list — chatops registry not loaded")
   469→        }
   470→        // Every entry must have a name + tier + permission (the fields the
   471→        // frontend renders). Don't assert the exact count — that would couple
   472→        // the test to the chatops command count, which drifts as commands are
   473→        // added.
   474→        for _, raw := range commands {
   475→                c, ok := raw.(map[string]any)
   476→                if !ok {
   477→                        t.Errorf("command entry is not an object: %T", raw)
   478→                        continue
   479→                }
   480→                if c["name"] == "" || c["name"] == nil {
   481→                        t.Error("command entry missing name")
   482→                }
   483→                if c["tier"] == "" || c["tier"] == nil {
   484→                        t.Error("command entry missing tier")
   485→                }
   486→                if c["permission"] == "" || c["permission"] == nil {
   487→                        t.Error("command entry missing permission")
   488→                }
   489→        }
   490→}
